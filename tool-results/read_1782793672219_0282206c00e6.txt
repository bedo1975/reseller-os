   595→  };
   596→  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
   597→  const url  = URL.createObjectURL(blob);
   598→  const a    = document.createElement('a');
   599→  a.href     = url;
   600→  a.download = `vinted-backup-${data._date}.json`;
   601→  a.click();
   602→  URL.revokeObjectURL(url);
   603→}

   605→function importData(event) {
   606→  const file = event.target.files[0];
   607→  if (!file) return;
   608→  const reader = new FileReader();
   609→  reader.onload = e => {
   610→    try {
   611→      const data = JSON.parse(e.target.result);
   612→      if (!data.vintedBrands && !data.vintedSearches) throw new Error('Fichier invalide');
   613→      if (!confirm('Importer ces données ? Tes données actuelles seront remplacées.')) return;
   614→      if (data.vintedBrands)      { localStorage.setItem('vintedBrands',      data.vintedBrands);      sbSave('vintedBrands',      data.vintedBrands); }
   615→      if (data.brandTags)         { localStorage.setItem('brandTags',         data.brandTags);         sbSave('brandTags',         data.brandTags); }
   616→      if (data.vintedCategories2) { localStorage.setItem('vintedCategories2', data.vintedCategories2); sbSave('vintedCategories2', data.vintedCategories2); }
   617→      if (data.vintedSearches)    { localStorage.setItem('vintedSearches',    data.vintedSearches);    sbSave('vintedSearches',    data.vintedSearches); }
   618→      alert('✅ Import réussi ! La page va se recharger.');
   619→      location.reload();
   620→    } catch(err) {
   621→      alert('❌ Fichier invalide ou corrompu.');
   622→    }
   623→  };
   624→  reader.readAsText(file);
   625→  event.target.value = ''; // reset pour pouvoir réimporter le même fichier
   626→}

   629→function initDragAndDrop() {
   630→  const container = document.getElementById('savedList');
   631→  let dragSrc = null;
   632→
   633→  container.addEventListener('dragstart', e => {
   634→    const item = e.target.closest('.search-item');
   635→    if (!item) return;
   636→    dragSrc = item;
   637→    e.dataTransfer.effectAllowed = 'move';
   638→    e.dataTransfer.setData('text/plain', item.dataset.name);
   639→    setTimeout(() => item.classList.add('dragging'), 0);
   640→  });
   641→
   642→  container.addEventListener('dragover', e => {
   643→    e.preventDefault();
   644→    e.dataTransfer.dropEffect = 'move';
   645→    const item = e.target.closest('.search-item');
   646→    container.querySelectorAll('.search-item').forEach(el => el.classList.remove('drag-over-top', 'drag-over-bottom'));
   647→    if (item && item !== dragSrc) {
   648→      const rect = item.getBoundingClientRect();
   649→      if (e.clientY < rect.top + rect.height / 2) {
   650→        item.classList.add('drag-over-top');
   651→      } else {
   652→        item.classList.add('drag-over-bottom');
   653→      }
   654→    }
   655→  });
   656→
   657→  container.addEventListener('dragleave', e => {
   658→    if (!container.contains(e.relatedTarget)) {
   659→      container.querySelectorAll('.search-item').forEach(el => el.classList.remove('drag-over-top', 'drag-over-bottom'));
   660→    }
   661→  });
   662→
   663→  container.addEventListener('drop', e => {
   664→    e.preventDefault();
   665→    const item = e.target.closest('.search-item');
   666→    if (!item || item === dragSrc) return;
   667→    const searches = JSON.parse(localStorage.getItem('vintedSearches') || '[]');
   668→    const srcIdx = searches.findIndex(s => s.name === dragSrc.dataset.name);
   669→    const tgtIdx = searches.findIndex(s => s.name === item.dataset.name);
   670→    const rect = item.getBoundingClientRect();
   671→    const insertBefore = e.clientY < rect.top + rect.height / 2;
   672→    const [moved] = searches.splice(srcIdx, 1);
   673→    const newTgt = searches.findIndex(s => s.name === item.dataset.name);
   674→    searches.splice(insertBefore ? newTgt : newTgt + 1, 0, moved);
   675→    const searchesStr = JSON.stringify(searches);
   676→    localStorage.setItem('vintedSearches', searchesStr);
   677→    sbSave('vintedSearches', searchesStr);
   678→    renderSavedSearches();
   679→  });
   680→
   681→  container.addEventListener('dragend', () => {
   682→    container.querySelectorAll('.search-item').forEach(el => el.classList.remove('dragging', 'drag-over-top', 'drag-over-bottom'));
   683→    dragSrc = null;
   684→  });
   685→}

   705→function getCatIds(key) {

   717→function buildCategories() {
   718→  const grid = document.getElementById('catGrid');
   719→  grid.innerHTML = '';
   720→  CATEGORIES.forEach((cat, i) => {
   721→    const hasChildren = cat.children && cat.children.length > 0;
   722→    if (!hasChildren) {
   723→      // Chip simple
   724→      const key = catKey(i);
   725→      const label = document.createElement('label');
   726→      label.className = 'cat-chip' + (selectedCatKeys.has(key) ? ' selected' : '');
   727→      label.innerHTML = `<input type="checkbox" data-key="${key}" ${selectedCatKeys.has(key) ? 'checked' : ''} onchange="toggleCat(this)"> ${cat.emoji} ${cat.label}`;
   728→      grid.appendChild(label);
   729→    } else {
   730→      // Chip avec enfants — on ne peut pas sélectionner le parent directement
   731→      selectedCatKeys.delete(catKey(i)); // nettoie toute sélection parente résiduelle
   732→      const wrapper = document.createElement('div');
   733→      wrapper.className = 'cat-expandable';
   734→      const anyChildSelected = cat.children.some((_, j) => selectedCatKeys.has(catKey(i, j)));
   735→      const isOpen = anyChildSelected; // Auto-déplier si des enfants sont sélectionnés
   736→      wrapper.innerHTML = `
   737→        <div class="cat-parent-chip ${anyChildSelected ? 'has-selected' : ''}" onclick="toggleExpand(this, ${i})">
   738→          <span class="cat-arrow ${isOpen ? 'open' : ''}">▶</span>
   739→          ${cat.emoji} ${cat.label}
   740→          ${cat.id && cat.id.length ? '' : ''}
   741→        </div>
   742→        <div class="cat-children ${isOpen ? 'open' : ''}" id="catChildren_${i}">
   743→          ${cat.children.map((child, j) => {
   744→            const key = catKey(i, j);
   745→            return `<label class="child-chip ${selectedCatKeys.has(key) ? 'selected' : ''}">
   746→              <input type="checkbox" data-key="${key}" ${selectedCatKeys.has(key) ? 'checked' : ''} onchange="toggleCat(this)">
   747→              ${child.emoji || ''} ${child.label}
   748→            </label>`;
   749→          }).join('')}
   750→        </div>`;
   751→      grid.appendChild(wrapper);
   752→    }
   753→  });
   754→}

   764→function toggleCat(cb) {
   765→  const key = cb.dataset.key;
   766→  if (cb.checked) selectedCatKeys.add(key);
   767→  else selectedCatKeys.delete(key);
   768→  cb.closest('label').classList.toggle('selected', cb.checked);
   769→  // Mettre à jour le style du parent si c'est un enfant
   770→  if (key.includes('-')) {
   771→    const parentIdx = key.split('-')[0];
   772→    const parentChip = document.querySelector(`.cat-expandable:nth-child(${+parentIdx + 1}) .cat-parent-chip`);
   773→    if (parentChip) {
   774→      const anySelected = CATEGORIES[parentIdx]?.children?.some((_, j) => selectedCatKeys.has(catKey(parentIdx, j)));
   775→      parentChip.classList.toggle('has-selected', !!anySelected);
   776→    }
   777→  }
   778→  updateCatCount();
   779→  updatePreview();
   780→}

   787→// Retourne un Map { brandId → [nomRecherche, ...] } pour toutes les recherches sauvegardées
   788→// excludeName : la recherche en cours d'édition — ses marques ne comptent pas comme "déjà utilisées"
   789→function getUsedBrandsMap(excludeName = null) {
   790→  const searches = JSON.parse(localStorage.getItem('vintedSearches') || '[]');
   791→  const map = new Map(); // id → [nom de recherche]
   792→  searches.forEach(s => {
   793→    if (s.name === excludeName) return; // ne pas compter la recherche en cours
   794→    (s.state?.brands || []).forEach(id => {
   795→      const numId = Number(id);
   796→      if (!map.has(numId)) map.set(numId, []);
   797→      map.get(numId).push(s.name);
   798→    });
   799→  });
   800→  return map;
   801→}

  1168→function getSearchDesc(state) {
  1169→  const catLabels = (state.cats || []).map(key => {
  1170→    const parts = key.split('-').map(Number);
  1171→    const cat   = CATEGORIES[parts[0]];
  1172→    if (!cat) return null;
  1173→    if (parts.length === 1) return cat.label;
  1174→    return cat.children?.[parts[1]]?.label || null;
  1175→  }).filter(Boolean);
  1176→  const parts = [];
  1177→  if (catLabels.length) parts.push(catLabels.join(', '));
  1178→  const nb = (state.brands || []).length;
  1179→  if (nb) parts.push(`${nb} marque${nb > 1 ? 's' : ''}`);
  1180→  const statuses = (state.status || []).map(v => STATUS_LABELS[v]).filter(Boolean);
  1181→  if (statuses.length) parts.push(statuses.join('+'));
  1182→  const sizes = (state.sizes || []).map(v => SIZE_LABELS[v]).filter(Boolean);
  1183→  if (sizes.length) parts.push(sizes.join('+'));
  1184→  const shoes = (state.shoes || []).map(v => SHOE_SIZE_LABELS[v]).filter(Boolean);
  1185→  if (shoes.length) parts.push('pt.' + shoes.join('+'));
  1186→  return parts.join(' · ') || 'Recherche vide';
  1187→}

  1260→function duplicateSearch(name) {
  1261→  const searches = JSON.parse(localStorage.getItem('vintedSearches') || '[]');
  1262→  const entry = searches.find(s => s.name === name);
  1263→  if (!entry) return;
  1264→  // Générer un nom unique
  1265→  let newName = name + ' (copie)';
  1266→  let i = 2;

  1750→function escHtml(s) {
  1751→  return String(s == null ? '' : s)
  1752→    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
  1753→    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');

  1763→// Reconstruit l'URL Vinted depuis un état sauvegardé (miroir de buildParams)
  1764→function buildParamsFromState(state) {
  1765→  const params = new URLSearchParams();
  1766→  const allCatIds = new Set();
  1767→  (state.cats || []).forEach(key => getCatIds(key).forEach(id => allCatIds.add(id)));
  1768→  allCatIds.forEach(id => params.append('catalog[]', id));
  1769→  (state.brands || []).forEach(id => params.append('brand_ids[]', id));
  1770→  (state.status || []).forEach(v => params.append('status_ids[]', v));
  1771→  (state.sizes  || []).forEach(v => params.append('size_ids[]', v));
  1772→  (state.shoes  || []).forEach(v => params.append('size_ids[]', v));
  1773→  if (state.priceMin) params.set('price_from', state.priceMin);
  1774→  if (state.priceMax) params.set('price_to', state.priceMax);
  1775→  params.set('order', state.order || 'newest_first');
  1776→  return params;
  1777→}

  1782→  if (!entry) return;

  1787→function catLabelsFromState(state) {
  1788→  return (state.cats || []).map(key => {
  1789→    const parts = key.split('-').map(Number);
  1790→    const cat = CATEGORIES[parts[0]];
  1791→    if (!cat) return null;

  1830→function renderSearchDetail(name) {
  1831→  const panel = document.getElementById('svDetail');
  1832→  const entry = getSearches().find(s => s.name === name);
  1833→  if (!entry) {
  1834→    panel.innerHTML = '<div class="sv-detail-empty">Sélectionne une recherche à gauche pour voir son détail.</div>';
  1835→    return;
  1836→  }
  1837→  const st = entry.state || {};
  1838→  const safeName = entry.name.replace(/\\/g, '\\\\').replace(/'/g, "\\'");
  1839→
  1840→  const brandMap = {};
  1841→  loadBrands().forEach(b => { brandMap[b.id] = b.name; });
  1842→  const brandNames = (st.brands || []).map(id => brandMap[id] || ('#' + id))
  1843→    .sort((a, b) => a.localeCompare(b, 'fr', { sensitivity: 'base' }));
  1844→  const brandsHtml = brandNames.length
  1845→    ? brandNames.map(n => `<span class="scard-chip">${escHtml(n)}</span>`).join('')
  1846→    : '<span class="scard-empty-crit">Aucune marque</span>';

  1848→  const crit = [];

  1850→  (st.status || []).map(v => STATUS_LABELS[v]).filter(Boolean).forEach(l => crit.push(`<span class="scard-chip crit">${escHtml(l)}</span>`));
  1851→  (st.sizes  || []).map(v => SIZE_LABELS[v]).filter(Boolean).forEach(l => crit.push(`<span class="scard-chip crit">${escHtml(l)}</span>`));
  1852→  (st.shoes  || []).map(v => SHOE_SIZE_LABELS[v]).filter(Boolean).forEach(l => crit.push(`<span class="scard-chip crit">pt. ${escHtml(l)}</span>`));
  1853→  if (st.priceMin || st.priceMax) {
  1854→    crit.push(`<span class="scard-chip crit">${escHtml((st.priceMin || '0') + '–' + (st.priceMax || '∞') + ' €')}</span>`);

  1857→    crit.push(`<span class="scard-chip crit">${escHtml(ORDER_LABELS[st.order])}</span>`);
  1858→  }
  1859→  const critHtml = crit.length ? crit.join('') : '<span class="scard-empty-crit">Aucun critère</span>';

  1861→  panel.innerHTML = `
  1862→    <div class="sv-detail-head">
  1863→      <div class="sv-detail-title">${escHtml(entry.name)}</div>
  1864→      <div class="sv-detail-actions">
  1865→        <button class="scard-open" onclick="openSearchByName('${safeName}')">▶ Voir sur Vinted</button>

  1870→      <div class="scard-label">Critères</div>

  1872→    </div>
  1873→    <div class="scard-block">
  1874→      <div class="scard-label">Marques (${brandNames.length})</div>
  1875→      <div class="scard-brands cols">${brandsHtml}</div>
  1876→    </div>`;
