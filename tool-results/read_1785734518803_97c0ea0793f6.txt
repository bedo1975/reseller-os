   410→   410→                const actions = permData[mod.key] || []
   411→   411→                const modActions = MODULE_ACTIONS_MAP[mod.key] || ['view']
   412→   412→                return (
   413→   413→                  <div key={mod.key} className="grid grid-cols-[1fr_repeat(7,auto)] gap-2 items-center px-2 py-1.5 rounded hover:bg-muted/30">

   441→   441→                    </div>

   447→   447→              <div className="flex gap-2 pt-3 border-t">

   453→   453→                  setPermData(all)

   455→   455→                  Tout autoriser
   456→   456→                </Button>

   458→   458→                  Tout révoquer
   459→   459→                </Button>

   461→   461→            </div>
   462→   462→          )}

   464→   464→          <DialogFooter className="gap-2">
   465→   465→            <Button variant="outline" onClick={() => setPermUser(null)}>Annuler</Button>

   469→   469→            </Button>

   471→   471→        </DialogContent>
   472→   472→      </Dialog>

   477→   477→function UserForm({ open, onOpenChange, user, onSaved }: {
   478→   478→  open: boolean
   479→   479→  onOpenChange: (o: boolean) => void
   480→   480→  user: UserRow | null
   481→   481→  onSaved: () => void
   482→   482→}) {
   483→   483→  const [saving, setSaving] = useState(false)
   484→   484→  const [form, setForm] = useState({
   485→   485→    name: '',
   486→   486→    email: '',
   487→   487→    password: '',
   488→   488→    role: 'staff' as 'admin' | 'staff',
   489→   489→  })
   490→   490→
   491→   491→  // Reset form when dialog opens
   492→   492→  useEffect(() => {
   493→   493→    if (open) {
   494→   494→      setForm({
   495→   495→        name: user?.name || '',
   496→   496→        email: user?.email || '',
   497→   497→        password: '',
   498→   498→        role: (user?.role as 'admin' | 'staff') || 'staff',
   499→   499→      })
   500→   500→    }
   501→   501→  }, [open, user])
   502→   502→
   503→   503→  const submit = async () => {
   504→   504→    if (!form.name) {
   505→   505→      toast.error('Nom requis')
   506→   506→      return
   507→   507→    }
   508→   508→    if (!user && !form.email) {
   509→   509→      toast.error('Email requis')
   510→   510→      return
   511→   511→    }
   512→   512→    if (!user && form.password.length < 8) {
   513→   513→      toast.error('Le mot de passe doit contenir au moins 8 caractères')
   514→   514→      return
   515→   515→    }
   516→   516→    if (user && form.password && form.password.length < 8) {
   517→   517→      toast.error('Le nouveau mot de passe doit contenir au moins 8 caractères')
   518→   518→      return
   519→   519→    }
   520→   520→
   521→   521→    setSaving(true)
   522→   522→    try {
   523→   523→      if (user) {
   524→   524→        // Edit mode
   525→   525→        const body: Record<string, unknown> = {
   526→   526→          name: form.name,
   527→   527→          role: form.role,
   528→   528→        }
   529→   529→        if (form.password) body.password = form.password
   530→   530→        const res = await fetch(`/api/users/${user.id}`, {
   531→   531→          method: 'PATCH',
   532→   532→          headers: { 'Content-Type': 'application/json' },
   533→   533→          body: JSON.stringify(body),
   534→   534→        })
   535→   535→        const data = await res.json()

   537→   537→        toast.success('Utilisateur modifié')

   540→   540→        const res = await fetch('/api/users', {
   541→   541→          method: 'POST',
   542→   542→          headers: { 'Content-Type': 'application/json' },
   543→   543→          body: JSON.stringify(form),
   544→   544→        })
   545→   545→        const data = await res.json()
   546→   546→        if (!res.ok) throw new Error(data.error || 'Erreur')
   547→   547→        toast.success('Utilisateur créé')
   548→   548→      }

   550→   550→    } catch (e: unknown) {
   551→   551→      toast.error(e instanceof Error ? e.message : 'Erreur')

   553→   553→      setSaving(false)

   564→   564→          </DialogTitle>
   565→   565→          <DialogDescription>
   566→   566→            {user
   567→   567→              ? `Modifiez le nom, le rôle ou réinitialisez le mot de passe de ${user.email}.`
   568→   568→              : 'Créez un nouveau compte Staff ou Admin.'}
   569→   569→          </DialogDescription>
   570→   570→        </DialogHeader>
   571→   571→        <div className="space-y-3 py-2">
   572→   572→          <div className="space-y-1.5">
   573→   573→            <Label className="text-xs">Nom complet *</Label>

   575→   575→              value={form.name}
   576→   576→              onChange={e => setForm({ ...form, name: e.target.value })}
   577→   577→              placeholder="Jean Dupont"
   578→   578→            />
   579→   579→          </div>
   580→   580→          <div className="space-y-1.5">
   581→   581→            <Label className="text-xs flex items-center gap-1.5">
