     1→'use client'
     2→
     3→import { useState, useMemo } from 'react'
     4→import { useFetch } from '@/hooks/use-fetch'
     5→import { useSettings } from '@/hooks/use-settings'
     6→import { Card, CardContent } from '@/components/ui/card'
     7→import { Button } from '@/components/ui/button'
     8→import { Input } from '@/components/ui/input'
     9→import { Label } from '@/components/ui/label'
    10→import { Badge } from '@/components/ui/badge'
    11→import {
    12→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    13→} from '@/components/ui/table'
    14→import {
    15→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    16→} from '@/components/ui/select'
    17→import {
    18→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription,
    19→} from '@/components/ui/dialog'
    20→import { Skeleton } from '@/components/ui/skeleton'
    21→import {
    22→  Plus, Search, MapPin, Barcode, Edit, Trash2, Package, ChevronLeft, ChevronRight,
    23→  Eye, AlertCircle,
    24→} from 'lucide-react'
    25→import { toast } from 'sonner'
    26→import {
    27→  formatEUR, formatDate, PUBLICATION_STATUSES,
    28→  getPubStatusColor, getPubStatusLabel,
    29→  getPlatformLabel, getPlatformColor, PLATFORMS,
    30→} from '@/lib/constants'
    31→import { cn } from '@/lib/utils'
    32→
    33→const PAGE_SIZE = 10
    34→
    35→interface Supplier {
    36→  id: string
    37→  name: string
    38→  type: string
    39→}
    40→
    41→interface StockItem {
    42→  id: string
    43→  sku: string
    44→  barcode: string | null
    45→  photos: string
    46→  brand: string
    47→  category: string
    48→  size: string | null
    49→  color: string | null
    50→  condition: string
    51→  purchaseDate: string
    52→  supplierId: string | null
    53→  supplier: Supplier | null
    54→  purchaseCost: number
    55→  lotReference: string | null
    56→  warehouse: string | null
    57→  rack: string | null
    58→  shelf: string | null
    59→  bin: string | null
    60→  status: string
    61→  platform: string | null
    62→  platforms: string  // JSON array
    63→  suggestedPrice: number | null
    64→  description: string | null
    65→  measurements: string | null
    66→  sale?: { id: string; salePrice: number; profit: number } | null
    67→}
    68→
    69→export function StockModule() {
    70→  const { data: items, loading, refresh } = useFetch<StockItem[]>('/api/stock')
    71→  const { data: suppliers } = useFetch<Supplier[]>('/api/suppliers')
    72→  const { getByType, getLabel } = useSettings()
    73→
    74→  const categories = getByType('category')
    75→  const conditions = getByType('condition')
    76→
    77→  const [search, setSearch] = useState('')
    78→  const [statusFilter, setStatusFilter] = useState<string>('all')
    79→  const [brandFilter, setBrandFilter] = useState<string>('all')
    80→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
    81→  const [page, setPage] = useState(1)
    82→  const [showForm, setShowForm] = useState(false)
    83→  const [editingItem, setEditingItem] = useState<StockItem | null>(null)
    84→  const [viewItem, setViewItem] = useState<StockItem | null>(null)
    85→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
    86→  const [showDeleteModal, setShowDeleteModal] = useState(false)
    87→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'single' | 'bulk'; item?: StockItem } | null>(null)
    88→
    89→  const brands = useMemo(() => {
    90→    if (!items) return []
    91→    return Array.from(new Set(items.map(i => i.brand))).sort()
    92→  }, [items])
    93→
    94→  const filtered = useMemo(() => {
    95→    if (!items) return []
    96→    return items.filter(i => {
    97→      if (statusFilter !== 'all' && i.status !== statusFilter) return false
    98→      if (brandFilter !== 'all' && i.brand !== brandFilter) return false
    99→      if (categoryFilter !== 'all' && i.category !== categoryFilter) return false
   100→      if (search) {
   101→        const q = search.toLowerCase()
   102→        return (
   103→          i.sku.toLowerCase().includes(q) ||
   104→          i.brand.toLowerCase().includes(q) ||
   105→          i.color?.toLowerCase().includes(q) ||
   106→          i.barcode?.toLowerCase().includes(q) ||
   107→          i.lotReference?.toLowerCase().includes(q)
   108→        )
   109→      }
   110→      return true
   111→    })
   112→  }, [items, search, statusFilter, brandFilter, categoryFilter])
   113→
   114→  // Quand les filtres changent, on reset la page via la clé de filtre
   115→  const filterKey = `${search}|${statusFilter}|${brandFilter}|${categoryFilter}`
   116→  const [lastFilterKey, setLastFilterKey] = useState(filterKey)
   117→  if (filterKey !== lastFilterKey) {
   118→    setLastFilterKey(filterKey)
   119→    setPage(1)
   120→  }
   121→
   122→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE))
   123→  const safePage = Math.min(page, totalPages)
   124→  const paginated = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE)
   125→
   126→  const totalStockValue = filtered.reduce((sum, i) => sum + i.purchaseCost, 0)
   127→  const totalPotentialRevenue = filtered.reduce((sum, i) => sum + (i.suggestedPrice || 0), 0)
   128→
   129→  // ─── Gestion de la sélection ───
   130→  const toggleSelect = (id: string) => {
   131→    setSelectedIds(prev => {
   132→      const next = new Set(prev)
   133→      if (next.has(id)) next.delete(id)
   134→      else next.add(id)
   135→      return next
   136→    })
   137→  }
   138→
   139→  const toggleSelectAll = () => {
   140→    if (selectedIds.size === paginated.length) {
   141→      // Tout est sélectionné sur cette page → désélectionner
   142→      setSelectedIds(new Set())
   143→    } else {
   144→      // Sinon → sélectionner tous les items de la page courante
   145→      setSelectedIds(new Set(paginated.map(i => i.id)))
   146→    }
   147→  }
   148→
   149→  const isAllSelected = paginated.length > 0 && paginated.every(i => selectedIds.has(i.id))
   150→  const isSomeSelected = paginated.some(i => selectedIds.has(i.id)) && !isAllSelected
   151→
   152→  // ─── Suppression (modale au lieu de confirm) ───
   153→  const askDeleteSingle = (item: StockItem) => {
   154→    setDeleteTarget({ type: 'single', item })
   155→    setShowDeleteModal(true)
   156→  }
   157→
   158→  const askDeleteBulk = () => {
   159→    if (selectedIds.size === 0) return
   160→    setDeleteTarget({ type: 'bulk' })
   161→    setShowDeleteModal(true)
   162→  }
   163→
   164→  const confirmDelete = async () => {
   165→    if (!deleteTarget) return
   166→    setShowDeleteModal(false)
   167→
   168→    if (deleteTarget.type === 'single' && deleteTarget.item) {
   169→      const res = await fetch(`/api/stock/${deleteTarget.item.id}`, { method: 'DELETE' })
   170→      if (res.ok) {
   171→        toast.success('Article supprimé')
   172→        refresh()
   173→      } else {
   174→        toast.error('Erreur lors de la suppression')
   175→      }
   176→    } else if (deleteTarget.type === 'bulk') {
   177→      // Suppression multiple
   178→      const ids = Array.from(selectedIds)
   179→      let okCount = 0
   180→      let errCount = 0
   181→      for (const id of ids) {
   182→        const res = await fetch(`/api/stock/${id}`, { method: 'DELETE' })
   183→        if (res.ok) okCount++
   184→        else errCount++
   185→      }
   186→      if (okCount > 0) toast.success(`${okCount} article${okCount > 1 ? 's' : ''} supprimé${okCount > 1 ? 's' : ''}`)
   187→      if (errCount > 0) toast.error(`${errCount} suppression(s) en échec`)
   188→      setSelectedIds(new Set())
   189→      refresh()
   190→    }
   191→    setDeleteTarget(null)
   192→  }
   193→
   194→  return (
   195→    <div className="space-y-5">
   196→      {/* Stats */}
   197→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   198→        <Card>
   199→          <CardContent className="p-4">
   200→            <p className="text-xs text-muted-foreground uppercase">Articles</p>
   201→            <p className="text-2xl font-bold mt-1">{filtered.length}</p>
   202→          </CardContent>
   203→        </Card>
   204→        <Card>
   205→          <CardContent className="p-4">
   206→            <p className="text-xs text-muted-foreground uppercase">Valeur stock</p>
   207→            <p className="text-2xl font-bold mt-1">{formatEUR(totalStockValue)}</p>
   208→          </CardContent>
   209→        </Card>
   210→        <Card>
   211→          <CardContent className="p-4">
   212→            <p className="text-xs text-muted-foreground uppercase">Potentiel vente</p>
   213→            <p className="text-2xl font-bold mt-1 text-emerald-600">{formatEUR(totalPotentialRevenue)}</p>
   214→          </CardContent>
   215→        </Card>
   216→        <Card>
   217→          <CardContent className="p-4">
   218→            <p className="text-xs text-muted-foreground uppercase">Marge potentielle</p>
   219→            <p className="text-2xl font-bold mt-1">
   220→              {totalStockValue > 0 ? Math.round(((totalPotentialRevenue - totalStockValue) / totalStockValue) * 100) : 0}%
   221→            </p>
   222→          </CardContent>
   223→        </Card>
   224→      </div>
   225→
   226→      {/* Toolbar */}
   227→      <Card>
   228→        <CardContent className="p-4">
   229→          <div className="flex flex-col lg:flex-row gap-3">
   230→            <div className="relative flex-1">
   231→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   232→              <Input
   233→                placeholder="Rechercher par SKU, marque, code-barres, lot..."
   234→                value={search}
   235→                onChange={e => setSearch(e.target.value)}
   236→                className="pl-9"
   237→              />
   238→            </div>
   239→            <div className="grid grid-cols-3 gap-2 lg:flex">
   240→              <Select value={statusFilter} onValueChange={setStatusFilter}>
   241→                <SelectTrigger className="w-full lg:w-[150px]">
   242→                  <SelectValue placeholder="Statut" />
   243→                </SelectTrigger>
   244→                <SelectContent>
   245→                  <SelectItem value="all">Tous statuts</SelectItem>
   246→                  {PUBLICATION_STATUSES.map(s => (
   247→                    <SelectItem key={s.id} value={s.id}>{s.label}</SelectItem>
   248→                  ))}
   249→                </SelectContent>
   250→              </Select>
   251→              <Select value={brandFilter} onValueChange={setBrandFilter}>
   252→                <SelectTrigger className="w-full lg:w-[140px]">
   253→                  <SelectValue placeholder="Marque" />
   254→                </SelectTrigger>
   255→                <SelectContent>
   256→                  <SelectItem value="all">Toutes marques</SelectItem>
   257→                  {brands.map(b => (
   258→                    <SelectItem key={b} value={b}>{b}</SelectItem>
   259→                  ))}
   260→                </SelectContent>
   261→              </Select>
   262→              <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   263→                <SelectTrigger className="w-full lg:w-[140px]">
   264→                  <SelectValue placeholder="Catégorie" />
   265→                </SelectTrigger>
   266→                <SelectContent>
   267→                  <SelectItem value="all">Toutes catégories</SelectItem>
   268→                  {categories.map(c => (
   269→                    <SelectItem key={c.id} value={c.code}>{c.value}</SelectItem>
   270→                  ))}
   271→                </SelectContent>
   272→              </Select>
   273→            </div>
   274→            <Button onClick={() => { setEditingItem(null); setShowForm(true) }}>
   275→              <Plus className="h-4 w-4 mr-2" /> Nouvel article
   276→            </Button>
   277→          </div>
   278→        </CardContent>
   279→      </Card>
   280→
   281→      {/* Barre d'actions bulk (visible quand des items sont sélectionnés) */}
   282→      {selectedIds.size > 0 && (
   283→        <Card className="border-emerald-300 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20">
   284→          <CardContent className="p-3 flex items-center justify-between gap-3">
   285→            <div className="flex items-center gap-3">
   286→              <div className="h-8 w-8 rounded-full bg-emerald-600 text-white flex items-center justify-center text-sm font-bold">
   287→                {selectedIds.size}
   288→              </div>
   289→              <span className="text-sm font-medium">
   290→                {selectedIds.size} article{selectedIds.size > 1 ? 's' : ''} sélectionné{selectedIds.size > 1 ? 's' : ''}
   291→              </span>
   292→            </div>
   293→            <div className="flex items-center gap-2">
   294→              <Button
   295→                variant="outline"
   296→                size="sm"
   297→                onClick={() => setSelectedIds(new Set())}
   298→              >
   299→                Désélectionner
   300→              </Button>
   301→              <Button
   302→                variant="destructive"
   303→                size="sm"
   304→                onClick={askDeleteBulk}
   305→              >
   306→                <Trash2 className="h-3.5 w-3.5 mr-1.5" />
   307→                Supprimer ({selectedIds.size})
   308→              </Button>
   309→            </div>
   310→          </CardContent>
   311→        </Card>
   312→      )}
   313→
   314→      {/* Table */}
   315→      <Card>
   316→        <CardContent className="p-0">
   317→          {loading ? (
   318→            <div className="p-4 space-y-2">
   319→              {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12" />)}
   320→            </div>
   321→          ) : filtered.length === 0 ? (
   322→            <div className="py-16 text-center">
   323→              <Package className="h-12 w-12 mx-auto text-muted-foreground mb-3" />
   324→              <p className="text-sm font-medium">Aucun article trouvé</p>
   325→              <p className="text-xs text-muted-foreground mt-1">Modifiez vos filtres ou ajoutez un nouvel article.</p>
   326→            </div>
   327→          ) : (
   328→            <>
   329→              <Table>
   330→                <TableHeader>
   331→                  <TableRow className="bg-muted/30">
   332→                    <TableHead className="w-10">
   333→                      <input
   334→                        type="checkbox"
   335→                        checked={isAllSelected}
   336→                        ref={el => { if (el) el.indeterminate = isSomeSelected }}
   337→                        onChange={toggleSelectAll}
   338→                        className="h-4 w-4 rounded border-border cursor-pointer"
   339→                        title="Tout sélectionner"
   340→                      />
   341→                    </TableHead>
   342→                    <TableHead className="w-12"></TableHead>
   343→                    <TableHead>SKU</TableHead>
   344→                    <TableHead>Marque</TableHead>
   345→                    <TableHead className="hidden md:table-cell">Catégorie</TableHead>
   346→                    <TableHead className="hidden lg:table-cell">Taille</TableHead>
   347→                    <TableHead className="hidden lg:table-cell">Couleur</TableHead>
   348→                    <TableHead className="hidden xl:table-cell">Emplacement</TableHead>
   349→                    <TableHead className="text-right">Coût</TableHead>
   350→                    <TableHead className="text-right hidden md:table-cell">Prix conseillé</TableHead>
   351→                    <TableHead>Statut</TableHead>
   352→                    <TableHead className="hidden lg:table-cell">Plateforme</TableHead>
   353→                    <TableHead className="text-right">Actions</TableHead>
   354→                  </TableRow>
   355→                </TableHeader>
   356→                <TableBody>
   357→                  {paginated.map(item => {
   358→                    const photos: string[] = (() => {
   359→                      try { return JSON.parse(item.photos) } catch { return [] }
   360→                    })()
   361→                    const location = [item.rack, item.shelf ? `Ét. ${item.shelf}` : null, item.bin ? `Bac ${item.bin}` : null].filter(Boolean).join(' · ')
   362→                    return (
   363→                      <TableRow
   364→                        key={item.id}
   365→                        className={cn(
   366→                          'hover:bg-muted/40 transition-colors',
   367→                          selectedIds.has(item.id) && 'bg-emerald-50/50 dark:bg-emerald-950/20'
   368→                        )}
   369→                      >
   370→                        <TableCell className="p-2" onClick={e => e.stopPropagation()}>
   371→                          <input
   372→                            type="checkbox"
   373→                            checked={selectedIds.has(item.id)}
   374→                            onChange={() => toggleSelect(item.id)}
   375→                            className="h-4 w-4 rounded border-border cursor-pointer"
   376→                          />
   377→                        </TableCell>
   378→                        <TableCell className="p-1.5 cursor-pointer" onClick={() => setViewItem(item)}>
   379→                          <div className="h-10 w-10 rounded-md overflow-hidden bg-muted shrink-0">
   380→                            {photos[0] ? (
   381→                              <img src={photos[0]} alt={item.brand} className="w-full h-full object-cover" />
   382→                            ) : (
   383→                              <div className="w-full h-full flex items-center justify-center">
   384→                                <Package className="h-4 w-4 text-muted-foreground/40" />
   385→                              </div>
   386→                            )}
   387→                          </div>
   388→                        </TableCell>
   389→                        <TableCell className="font-mono text-xs">{item.sku}</TableCell>
   390→                        <TableCell className="font-medium">{item.brand}</TableCell>
   391→                        <TableCell className="hidden md:table-cell text-muted-foreground text-xs">{getLabel('category', item.category)}</TableCell>
   392→                        <TableCell className="hidden lg:table-cell text-xs">{item.size || '—'}</TableCell>
   393→                        <TableCell className="hidden lg:table-cell text-xs">{item.color || '—'}</TableCell>
   394→                        <TableCell className="hidden xl:table-cell text-xs text-muted-foreground">
   395→                          {location ? (
   396→                            <span className="flex items-center gap-1">
   397→                              <MapPin className="h-3 w-3 shrink-0" /> {location}
   398→                            </span>
   399→                          ) : '—'}
   400→                        </TableCell>
   401→                        <TableCell className="text-right font-medium">{formatEUR(item.purchaseCost)}</TableCell>
   402→                        <TableCell className="text-right hidden md:table-cell">
   403→                          {item.suggestedPrice ? (
   404→                            <span className="font-semibold text-emerald-600">{formatEUR(item.suggestedPrice)}</span>
   405→                          ) : '—'}
   406→                        </TableCell>
   407→                        <TableCell>
   408→                          <span className={cn('text-[10px] font-semibold px-2 py-0.5 rounded-full inline-block', getPubStatusColor(item.status))}>
   409→                            {getPubStatusLabel(item.status)}
   410→                          </span>
   411→                        </TableCell>
   412→                        <TableCell className="hidden lg:table-cell">
   413→                          {item.platform ? (
   414→                            <span className={cn('text-[10px] font-semibold px-2 py-0.5 rounded-full inline-block', getPlatformColor(item.platform))}>
   415→                              {getPlatformLabel(item.platform)}
   416→                            </span>
   417→                          ) : '—'}
   418→                        </TableCell>
   419→                        <TableCell className="text-right" onClick={e => e.stopPropagation()}>
   420→                          <div className="flex items-center justify-end gap-1">
   421→                            <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => setViewItem(item)}>
   422→                              <Eye className="h-3.5 w-3.5" />
   423→                            </Button>
   424→                            <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => { setEditingItem(item); setShowForm(true) }}>
   425→                              <Edit className="h-3.5 w-3.5" />
   426→                            </Button>
   427→                            <Button variant="ghost" size="sm" className="h-7 w-7 p-0 text-rose-600 hover:text-rose-700" onClick={() => askDeleteSingle(item)}>
   428→                              <Trash2 className="h-3.5 w-3.5" />
   429→                            </Button>
   430→                          </div>
   431→                        </TableCell>
   432→                      </TableRow>
   433→                    )
   434→                  })}
   435→                </TableBody>
   436→              </Table>
   437→
   438→              {/* Pagination */}
   439→              <div className="flex items-center justify-between px-4 py-3 border-t">
   440→                <p className="text-xs text-muted-foreground">
   441→                  {filtered.length} article{filtered.length > 1 ? 's' : ''} ·{' '}
   442→                  Page {safePage} sur {totalPages} ·{' '}
   443→                  Affiche {(safePage - 1) * PAGE_SIZE + 1}–{Math.min(safePage * PAGE_SIZE, filtered.length)}
   444→                </p>
   445→                <div className="flex items-center gap-1">
   446→                  <Button
   447→                    variant="outline"
   448→                    size="sm"
   449→                    disabled={safePage === 1}
   450→                    onClick={() => setPage(p => Math.max(1, p - 1))}
   451→                    className="h-8"
   452→                  >
   453→                    <ChevronLeft className="h-3.5 w-3.5 mr-1" /> Préc.
   454→                  </Button>
   455→                  {Array.from({ length: totalPages }, (_, i) => i + 1)
   456→                    .filter(p => p === 1 || p === totalPages || Math.abs(p - safePage) <= 1)
   457→                    .map((p, idx, arr) => {
   458→                      const showEllipsisBefore = idx > 0 && arr[idx - 1] !== p - 1
   459→                      return (
   460→                        <span key={p} className="flex items-center">
   461→                          {showEllipsisBefore && <span className="px-1 text-muted-foreground">…</span>}
   462→                          <Button
   463→                            variant={p === safePage ? 'default' : 'outline'}
   464→                            size="sm"
   465→                            className="h-8 w-8 p-0"
   466→                            onClick={() => setPage(p)}
   467→                          >
   468→                            {p}
   469→                          </Button>
   470→                        </span>
   471→                      )
   472→                    })}
   473→                  <Button
   474→                    variant="outline"
   475→                    size="sm"
   476→                    disabled={safePage === totalPages}
   477→                    onClick={() => setPage(p => Math.min(totalPages, p + 1))}
   478→                    className="h-8"
   479→                  >
   480→                    Suiv. <ChevronRight className="h-3.5 w-3.5 ml-1" />
   481→                  </Button>
   482→                </div>
   483→              </div>
   484→            </>
   485→          )}
   486→        </CardContent>
   487→      </Card>
   488→
   489→      {/* Form Dialog */}
   490→      <StockForm
   491→        open={showForm}
   492→        onOpenChange={setShowForm}
   493→        item={editingItem}
   494→        suppliers={suppliers || []}
   495→        categories={categories}
   496→        conditions={conditions}
   497→        sizes={getByType('size')}
   498→        colors={getByType('color')}
   499→        onSaved={() => { setShowForm(false); refresh() }}
   500→      />
   501→
   502→      {/* View Dialog */}
   503→      <StockDetail open={!!viewItem} onOpenChange={(o) => !o && setViewItem(null)} item={viewItem} />
   504→
   505→      {/* Modale de confirmation de suppression */}
   506→      <Dialog open={showDeleteModal} onOpenChange={(o) => { setShowDeleteModal(o); if (!o) setDeleteTarget(null) }}>
   507→        <DialogContent className="max-w-md">
   508→          <DialogHeader>
   509→            <DialogTitle className="flex items-center gap-2 text-rose-700 dark:text-rose-300">
   510→              <AlertCircle className="h-5 w-5" />

   556→              <Trash2 className="h-4 w-4 mr-1.5" />

   558→                ? `Supprimer ${selectedIds.size} article${selectedIds.size > 1 ? 's' : ''}`
   559→                : 'Supprimer'
   560→              }
   561→            </Button>
   562→          </DialogFooter>
   563→        </DialogContent>
   564→      </Dialog>
   565→    </div>
   566→  )
