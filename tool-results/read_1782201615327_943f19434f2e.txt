    68→    68→
    69→    69→export function StockModule() {
    70→    70→  const { data: items, loading, refresh } = useFetch<StockItem[]>('/api/stock')
    71→    71→  const { data: suppliers } = useFetch<Supplier[]>('/api/suppliers')
    72→    72→  const { getByType, getLabel } = useSettings()
    73→    73→
    74→    74→  const categories = getByType('category')
    75→    75→  const conditions = getByType('condition')
    76→    76→
    77→    77→  const [search, setSearch] = useState('')
    78→    78→  const [statusFilter, setStatusFilter] = useState<string>('all')
    79→    79→  const [brandFilter, setBrandFilter] = useState<string>('all')
    80→    80→  const [categoryFilter, setCategoryFilter] = useState<string>('all')
    81→    81→  const [page, setPage] = useState(1)
    82→    82→  const [showForm, setShowForm] = useState(false)
    83→    83→  const [editingItem, setEditingItem] = useState<StockItem | null>(null)
    84→    84→  const [viewItem, setViewItem] = useState<StockItem | null>(null)
    85→    85→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
    86→    86→  const [showDeleteModal, setShowDeleteModal] = useState(false)
    87→    87→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'single' | 'bulk'; item?: StockItem } | null>(null)
    88→    88→
    89→    89→  const brands = useMemo(() => {
    90→    90→    if (!items) return []
    91→    91→    return Array.from(new Set(items.map(i => i.brand))).sort()
    92→    92→  }, [items])
    93→    93→
    94→    94→  const filtered = useMemo(() => {
    95→    95→    if (!items) return []
    96→    96→    return items.filter(i => {
    97→    97→      if (statusFilter !== 'all' && i.status !== statusFilter) return false
    98→    98→      if (brandFilter !== 'all' && i.brand !== brandFilter) return false
    99→    99→      if (categoryFilter !== 'all' && i.category !== categoryFilter) return false
   100→   100→      if (search) {
   101→   101→        const q = search.toLowerCase()
   102→   102→        return (
   103→   103→          i.sku.toLowerCase().includes(q) ||
   104→   104→          i.brand.toLowerCase().includes(q) ||
   105→   105→          i.color?.toLowerCase().includes(q) ||
   106→   106→          i.barcode?.toLowerCase().includes(q) ||
   107→   107→          i.lotReference?.toLowerCase().includes(q)
   108→   108→        )
   109→   109→      }
   110→   110→      return true
   111→   111→    })
   112→   112→  }, [items, search, statusFilter, brandFilter, categoryFilter])
   113→   113→
   114→   114→  // Quand les filtres changent, on reset la page via la clé de filtre
   115→   115→  const filterKey = `${search}|${statusFilter}|${brandFilter}|${categoryFilter}`
   116→   116→  const [lastFilterKey, setLastFilterKey] = useState(filterKey)
   117→   117→  if (filterKey !== lastFilterKey) {
   118→   118→    setLastFilterKey(filterKey)
   119→   119→    setPage(1)
   120→   120→  }
   121→   121→
   122→   122→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE))
   123→   123→  const safePage = Math.min(page, totalPages)
   124→   124→  const paginated = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE)
   125→   125→
   126→   126→  const totalStockValue = filtered.reduce((sum, i) => sum + i.purchaseCost, 0)
   127→   127→  const totalPotentialRevenue = filtered.reduce((sum, i) => sum + (i.suggestedPrice || 0), 0)
   128→   128→
   129→   129→  // ─── Gestion de la sélection ───
   130→   130→  const toggleSelect = (id: string) => {
   131→   131→    setSelectedIds(prev => {
   132→   132→      const next = new Set(prev)
   133→   133→      if (next.has(id)) next.delete(id)
   134→   134→      else next.add(id)
   135→   135→      return next
   136→   136→    })
   137→   137→  }
   138→   138→
   139→   139→  const toggleSelectAll = () => {
   140→   140→    if (selectedIds.size === paginated.length) {
   141→   141→      // Tout est sélectionné sur cette page → désélectionner
   142→   142→      setSelectedIds(new Set())
   143→   143→    } else {
   144→   144→      // Sinon → sélectionner tous les items de la page courante
   145→   145→      setSelectedIds(new Set(paginated.map(i => i.id)))
   146→   146→    }
   147→   147→  }
   148→   148→
   149→   149→  const isAllSelected = paginated.length > 0 && paginated.every(i => selectedIds.has(i.id))
   150→   150→  const isSomeSelected = paginated.some(i => selectedIds.has(i.id)) && !isAllSelected
   151→   151→
   152→   152→  // ─── Suppression (modale au lieu de confirm) ───
   153→   153→  const askDeleteSingle = (item: StockItem) => {
   154→   154→    setDeleteTarget({ type: 'single', item })
   155→   155→    setShowDeleteModal(true)
   156→   156→  }
   157→   157→
   158→   158→  const askDeleteBulk = () => {
   159→   159→    if (selectedIds.size === 0) return
   160→   160→    setDeleteTarget({ type: 'bulk' })
   161→   161→    setShowDeleteModal(true)
   162→   162→  }
   163→   163→
   164→   164→  const confirmDelete = async () => {
   165→   165→    if (!deleteTarget) return
   166→   166→    setShowDeleteModal(false)
   167→   167→
   168→   168→    if (deleteTarget.type === 'single' && deleteTarget.item) {
   169→   169→      const res = await fetch(`/api/stock/${deleteTarget.item.id}`, { method: 'DELETE' })
   170→   170→      if (res.ok) {
   171→   171→        toast.success('Article supprimé')
   172→   172→        refresh()
   173→   173→      } else {
   174→   174→        toast.error('Erreur lors de la suppression')
   175→   175→      }
   176→   176→    } else if (deleteTarget.type === 'bulk') {
   177→   177→      // Suppression multiple
   178→   178→      const ids = Array.from(selectedIds)
   179→   179→      let okCount = 0
   180→   180→      let errCount = 0
   181→   181→      for (const id of ids) {
   182→   182→        const res = await fetch(`/api/stock/${id}`, { method: 'DELETE' })
   183→   183→        if (res.ok) okCount++
   184→   184→        else errCount++
   185→   185→      }
   186→   186→      if (okCount > 0) toast.success(`${okCount} article${okCount > 1 ? 's' : ''} supprimé${okCount > 1 ? 's' : ''}`)
   187→   187→      if (errCount > 0) toast.error(`${errCount} suppression(s) en échec`)
   188→   188→      setSelectedIds(new Set())
   189→   189→      refresh()
   190→   190→    }
   191→   191→    setDeleteTarget(null)
   192→   192→  }
   193→   193→
   194→   194→  return (
   195→   195→    <div className="space-y-5">
   196→   196→      {/* Stats */}
   197→   197→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   198→   198→        <Card>
   199→   199→          <CardContent className="p-4">
   200→   200→            <p className="text-xs text-muted-foreground uppercase">Articles</p>
   201→   201→            <p className="text-2xl font-bold mt-1">{filtered.length}</p>
   202→   202→          </CardContent>
   203→   203→        </Card>
   204→   204→        <Card>
   205→   205→          <CardContent className="p-4">
   206→   206→            <p className="text-xs text-muted-foreground uppercase">Valeur stock</p>
   207→   207→            <p className="text-2xl font-bold mt-1">{formatEUR(totalStockValue)}</p>
   208→   208→          </CardContent>
   209→   209→        </Card>
   210→   210→        <Card>
   211→   211→          <CardContent className="p-4">
   212→   212→            <p className="text-xs text-muted-foreground uppercase">Potentiel vente</p>
   213→   213→            <p className="text-2xl font-bold mt-1 text-emerald-600">{formatEUR(totalPotentialRevenue)}</p>
   214→   214→          </CardContent>
   215→   215→        </Card>
   216→   216→        <Card>
   217→   217→          <CardContent className="p-4">
   218→   218→            <p className="text-xs text-muted-foreground uppercase">Marge potentielle</p>
   219→   219→            <p className="text-2xl font-bold mt-1">
   220→   220→              {totalStockValue > 0 ? Math.round(((totalPotentialRevenue - totalStockValue) / totalStockValue) * 100) : 0}%
   221→   221→            </p>
   222→   222→          </CardContent>
   223→   223→        </Card>
   224→   224→      </div>
   225→   225→
   226→   226→      {/* Toolbar */}
   227→   227→      <Card>
   228→   228→        <CardContent className="p-4">
   229→   229→          <div className="flex flex-col lg:flex-row gap-3">
   230→   230→            <div className="relative flex-1">
   231→   231→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   232→   232→              <Input
   233→   233→                placeholder="Rechercher par SKU, marque, code-barres, lot..."
   234→   234→                value={search}
   235→   235→                onChange={e => setSearch(e.target.value)}
   236→   236→                className="pl-9"
   237→   237→              />
   238→   238→            </div>
   239→   239→            <div className="grid grid-cols-3 gap-2 lg:flex">
   240→   240→              <Select value={statusFilter} onValueChange={setStatusFilter}>
   241→   241→                <SelectTrigger className="w-full lg:w-[150px]">
   242→   242→                  <SelectValue placeholder="Statut" />
   243→   243→                </SelectTrigger>
   244→   244→                <SelectContent>
   245→   245→                  <SelectItem value="all">Tous statuts</SelectItem>
   246→   246→                  {PUBLICATION_STATUSES.map(s => (
   247→   247→                    <SelectItem key={s.id} value={s.id}>{s.label}</SelectItem>
   248→   248→                  ))}
   249→   249→                </SelectContent>
   250→   250→              </Select>
   251→   251→              <Select value={brandFilter} onValueChange={setBrandFilter}>
   252→   252→                <SelectTrigger className="w-full lg:w-[140px]">
   253→   253→                  <SelectValue placeholder="Marque" />
   254→   254→                </SelectTrigger>
   255→   255→                <SelectContent>
   256→   256→                  <SelectItem value="all">Toutes marques</SelectItem>
   257→   257→                  {brands.map(b => (
   258→   258→                    <SelectItem key={b} value={b}>{b}</SelectItem>
   259→   259→                  ))}
   260→   260→                </SelectContent>
   261→   261→              </Select>
   262→   262→              <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   263→   263→                <SelectTrigger className="w-full lg:w-[140px]">
   264→   264→                  <SelectValue placeholder="Catégorie" />
   265→   265→                </SelectTrigger>
   266→   266→                <SelectContent>
   267→   267→                  <SelectItem value="all">Toutes catégories</SelectItem>
   268→   268→                  {categories.map(c => (
   269→   269→                    <SelectItem key={c.id} value={c.code}>{c.value}</SelectItem>
   270→   270→                  ))}
   271→   271→                </SelectContent>
   272→   272→              </Select>
   273→   273→            </div>
   274→   274→            <Button onClick={() => { setEditingItem(null); setShowForm(true) }}>
   275→   275→              <Plus className="h-4 w-4 mr-2" /> Nouvel article
   276→   276→            </Button>
   277→   277→          </div>
   278→   278→        </CardContent>
   279→   279→      </Card>
   280→   280→
   281→   281→      {/* Barre d'actions bulk (visible quand des items sont sélectionnés) */}
   282→   282→      {selectedIds.size > 0 && (
   283→   283→        <Card className="border-emerald-300 dark:border-emerald-800 bg-emerald-50/50 dark:bg-emerald-950/20">
   284→   284→          <CardContent className="p-3 flex items-center justify-between gap-3">
   285→   285→            <div className="flex items-center gap-3">
   286→   286→              <div className="h-8 w-8 rounded-full bg-emerald-600 text-white flex items-center justify-center text-sm font-bold">
   287→   287→                {selectedIds.size}
   288→   288→              </div>
   289→   289→              <span className="text-sm font-medium">
   290→   290→                {selectedIds.size} article{selectedIds.size > 1 ? 's' : ''} sélectionné{selectedIds.size > 1 ? 's' : ''}
   291→   291→              </span>
   292→   292→            </div>
   293→   293→            <div className="flex items-center gap-2">
   294→   294→              <Button
   295→   295→                variant="outline"
   296→   296→                size="sm"
   297→   297→                onClick={() => setSelectedIds(new Set())}
   298→   298→              >
   299→   299→                Désélectionner
   300→   300→              </Button>
   301→   301→              <Button
   302→   302→                variant="destructive"
   303→   303→                size="sm"
   304→   304→                onClick={askDeleteBulk}
   305→   305→              >
   306→   306→                <Trash2 className="h-3.5 w-3.5 mr-1.5" />
   307→   307→                Supprimer ({selectedIds.size})
   308→   308→              </Button>
   309→   309→            </div>
   310→   310→          </CardContent>
   311→   311→        </Card>
   312→   312→      )}
   313→   313→
   314→   314→      {/* Table */}
   315→   315→      <Card>
   316→   316→        <CardContent className="p-0">
   317→   317→          {loading ? (
   318→   318→            <div className="p-4 space-y-2">
   319→   319→              {Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-12" />)}
   320→   320→            </div>
   321→   321→          ) : filtered.length === 0 ? (
   322→   322→            <div className="py-16 text-center">
   323→   323→              <Package className="h-12 w-12 mx-auto text-muted-foreground mb-3" />
   324→   324→              <p className="text-sm font-medium">Aucun article trouvé</p>
   325→   325→              <p className="text-xs text-muted-foreground mt-1">Modifiez vos filtres ou ajoutez un nouvel article.</p>
   326→   326→            </div>
   327→   327→          ) : (
   328→   328→            <>
   329→   329→              <Table>
   330→   330→                <TableHeader>
   331→   331→                  <TableRow className="bg-muted/30">
   332→   332→                    <TableHead className="w-10">
   333→   333→                      <input
   334→   334→                        type="checkbox"
   335→   335→                        checked={isAllSelected}
   336→   336→                        ref={el => { if (el) el.indeterminate = isSomeSelected }}
   337→   337→                        onChange={toggleSelectAll}
   338→   338→                        className="h-4 w-4 rounded border-border cursor-pointer"
   339→   339→                        title="Tout sélectionner"
   340→   340→                      />
   341→   341→                    </TableHead>
   342→   342→                    <TableHead className="w-12"></TableHead>
   343→   343→                    <TableHead>SKU</TableHead>
   344→   344→                    <TableHead>Marque</TableHead>
   345→   345→                    <TableHead className="hidden md:table-cell">Catégorie</TableHead>
   346→   346→                    <TableHead className="hidden lg:table-cell">Taille</TableHead>
   347→   347→                    <TableHead className="hidden lg:table-cell">Couleur</TableHead>
   348→   348→                    <TableHead className="hidden xl:table-cell">Emplacement</TableHead>
   349→   349→                    <TableHead className="text-right">Coût</TableHead>
   350→   350→                    <TableHead className="text-right hidden md:table-cell">Prix conseillé</TableHead>
   351→   351→                    <TableHead>Statut</TableHead>
   352→   352→                    <TableHead className="hidden lg:table-cell">Plateforme</TableHead>
   353→   353→                    <TableHead className="text-right">Actions</TableHead>
   354→   354→                  </TableRow>
   355→   355→                </TableHeader>
   356→   356→                <TableBody>
   357→   357→                  {paginated.map(item => {
   358→   358→                    const photos: string[] = (() => {
   359→   359→                      try { return JSON.parse(item.photos) } catch { return [] }
   360→   360→                    })()
   361→   361→                    const location = [item.rack, item.shelf ? `Ét. ${item.shelf}` : null, item.bin ? `Bac ${item.bin}` : null].filter(Boolean).join(' · ')
   362→   362→                    return (
   363→   363→                      <TableRow
   364→   364→                        key={item.id}
   365→   365→                        className={cn(
   366→   366→                          'hover:bg-muted/40 transition-colors',
   367→   367→                          selectedIds.has(item.id) && 'bg-emerald-50/50 dark:bg-emerald-950/20'
   368→   368→                        )}
   369→   369→                      >
   370→   370→                        <TableCell className="p-2" onClick={e => e.stopPropagation()}>
   371→   371→                          <input
   372→   372→                            type="checkbox"
   373→   373→                            checked={selectedIds.has(item.id)}
   374→   374→                            onChange={() => toggleSelect(item.id)}
   375→   375→                            className="h-4 w-4 rounded border-border cursor-pointer"
   376→   376→                          />
   377→   377→                        </TableCell>
   378→   378→                        <TableCell className="p-1.5 cursor-pointer" onClick={() => setViewItem(item)}>
   379→   379→                          <div className="h-10 w-10 rounded-md overflow-hidden bg-muted shrink-0">
   380→   380→                            {photos[0] ? (
   381→   381→                              <img src={photos[0]} alt={item.brand} className="w-full h-full object-cover" />
   382→   382→                            ) : (
   383→   383→                              <div className="w-full h-full flex items-center justify-center">
   384→   384→                                <Package className="h-4 w-4 text-muted-foreground/40" />
   385→   385→                              </div>
   386→   386→                            )}
   387→   387→                          </div>
   388→   388→                        </TableCell>
   389→   389→                        <TableCell className="font-mono text-xs">{item.sku}</TableCell>
   390→   390→                        <TableCell className="font-medium">{item.brand}</TableCell>
   391→   391→                        <TableCell className="hidden md:table-cell text-muted-foreground text-xs">{getLabel('category', item.category)}</TableCell>
   392→   392→                        <TableCell className="hidden lg:table-cell text-xs">{item.size || '—'}</TableCell>
   393→   393→                        <TableCell className="hidden lg:table-cell text-xs">{item.color || '—'}</TableCell>
   394→   394→                        <TableCell className="hidden xl:table-cell text-xs text-muted-foreground">
   395→   395→                          {location ? (
   396→   396→                            <span className="flex items-center gap-1">
   397→   397→                              <MapPin className="h-3 w-3 shrink-0" /> {location}
   398→   398→                            </span>
   399→   399→                          ) : '—'}
   400→   400→                        </TableCell>
   401→   401→                        <TableCell className="text-right font-medium">{formatEUR(item.purchaseCost)}</TableCell>
   402→   402→                        <TableCell className="text-right hidden md:table-cell">
   403→   403→                          {item.suggestedPrice ? (
   404→   404→                            <span className="font-semibold text-emerald-600">{formatEUR(item.suggestedPrice)}</span>
   405→   405→                          ) : '—'}
   406→   406→                        </TableCell>
   407→   407→                        <TableCell>
   408→   408→                          <span className={cn('text-[10px] font-semibold px-2 py-0.5 rounded-full inline-block', getPubStatusColor(item.status))}>
   409→   409→                            {getPubStatusLabel(item.status)}
   410→   410→                          </span>
   411→   411→                        </TableCell>
   412→   412→                        <TableCell className="hidden lg:table-cell">
   413→   413→                          {item.platform ? (
   414→   414→                            <span className={cn('text-[10px] font-semibold px-2 py-0.5 rounded-full inline-block', getPlatformColor(item.platform))}>
   415→   415→                              {getPlatformLabel(item.platform)}
   416→   416→                            </span>
   417→   417→                          ) : '—'}
   418→   418→                        </TableCell>
   419→   419→                        <TableCell className="text-right" onClick={e => e.stopPropagation()}>
   420→   420→                          <div className="flex items-center justify-end gap-1">
   421→   421→                            <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => setViewItem(item)}>
   422→   422→                              <Eye className="h-3.5 w-3.5" />
   423→   423→                            </Button>
   424→   424→                            <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => { setEditingItem(item); setShowForm(true) }}>
