    89→function fmtDate(iso: string): string {
    90→  try { return new Date(iso).toLocaleDateString('fr-FR') } catch { return '—' }
    91→}
    92→
    93→function fmtMoney(n: number): string {
    94→  return (n || 0).toFixed(2) + ' €'
    95→}

    97→function parseItems(json: string): PreOrderItem[] {
    98→  try { return JSON.parse(json) } catch { return [] }

   302→function CreatePreOrderForm({ onBack, onCreated }: { onBack: () => void; onCreated: () => void }) {
   303→  const { data: suppliers } = useFetch<Supplier[]>('/api/suppliers')
   304→  const { data: stockItems } = useFetch<StockItemLite[]>('/api/stock')
   305→  const { getByType, getSubcategories } = useSettings()
   306→  const sizes = getByType('size')
   307→  const colors = getByType('color')
   308→  const conditions = getByType('condition')
   309→  const categories = getByType('category')
   310→  const [saving, setSaving] = useState(false)
   311→  const [creatingArticleIdx, setCreatingArticleIdx] = useState<number | null>(null)
   312→  const [pickerIdx, setPickerIdx] = useState<number | null>(null)  // which item line is picking a product
   313→
   314→  const [name, setName] = useState('')
   315→  const [supplierId, setSupplierId] = useState<string>('')
   316→  const [orderDate, setOrderDate] = useState(new Date().toISOString().slice(0, 10))
   317→  const [paymentMethod, setPaymentMethod] = useState<string>('')
   318→  const [notes, setNotes] = useState('')
   319→  const [shippingCost, setShippingCost] = useState(0)
   320→  const [items, setItems] = useState<PreOrderItem[]>([
   321→    { designation: '', url: '', description: '', size: '', color: '', condition: '', quantity: 1, unitPrice: 0, stockItemId: null },
   322→  ])
   323→
   324→  const subtotal = useMemo(() => {
   325→    return items.reduce((s, it) => s + (Number(it.quantity) || 0) * (Number(it.unitPrice) || 0), 0)
   326→  }, [items])
   327→  const total = subtotal + (Number(shippingCost) || 0)
   328→
   329→  const updateItem = (idx: number, field: keyof PreOrderItem, value: any) => {
   330→    setItems(prev => prev.map((it, i) => i === idx ? { ...it, [field]: value } : it))
   331→  }
   332→
   333→  const addItem = () => {
   334→    setItems(prev => [...prev, { designation: '', url: '', description: '', size: '', color: '', condition: '', quantity: 1, unitPrice: 0, stockItemId: null }])
   335→  }
   336→
   337→  const removeItem = (idx: number) => {
   338→    setItems(prev => prev.length > 1 ? prev.filter((_, i) => i !== idx) : prev)
   339→  }
   340→
   341→  // When selecting an existing stock item, auto-fill the fields
   342→  const selectStockItem = (idx: number, stockItemId: string) => {
   343→    if (!stockItemId) {
   344→      updateItem(idx, 'stockItemId', null)
   345→      return
   346→    }
   347→    const item = (stockItems || []).find(i => i.id === stockItemId)
   348→    if (!item) return
   349→    setItems(prev => prev.map((it, i) => {
   350→      if (i !== idx) return it
   351→      const designation = item.title || `${item.brand} ${item.category}`.trim()
   352→      return {
   353→        ...it,
   354→        stockItemId: item.id,
   355→        designation,
   356→        size: item.size || '',
   357→        color: item.color || '',
   358→        condition: item.condition || '',
   359→      }
   360→    }))
   361→  }
   362→
   363→  const submit = async () => {
   364→    if (!name.trim()) { toast.error('Le nom est requis'); return }
   365→    if (items.length === 0 || items.every(i => !i.designation.trim())) {
   366→      toast.error('Au moins un article avec désignation est requis'); return
   367→    }
   368→    // Filter out empty items
   369→    const cleanItems = items.filter(i => i.designation.trim())
   370→    if (cleanItems.length === 0) { toast.error('Au moins un article est requis'); return }
   371→
   372→    setSaving(true)
   373→    try {
   374→      const res = await fetch('/api/preorders', {
   375→        method: 'POST',
   376→        headers: { 'Content-Type': 'application/json' },
   377→        body: JSON.stringify({
   378→          name: name.trim(),
   379→          supplierId: supplierId || null,
   380→          supplierName: supplierId ? (suppliers || []).find(s => s.id === supplierId)?.name : null,
   381→          orderDate,
   382→          paymentMethod: paymentMethod || null,
   383→          items: cleanItems,
   384→          shippingCost: Number(shippingCost) || 0,
   385→          notes: notes.trim() || null,
   386→        }),
   387→      })
   388→      const data = await res.json()
   389→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }
   390→      toast.success('Pré-commande créée')
   391→      onCreated()
   392→    } catch {
   393→      toast.error('Erreur réseau')
   394→    } finally {
   395→      setSaving(false)
   396→    }
   397→  }
   398→
   399→  // Create a StockItem from the current article line (qty = 0, status = A_PHOTOGRAPHIER)
   400→  // and link it back to the pre-order item.
   401→  //
   402→  // IMPORTANT: purchaseCost is set to 0 (not item.unitPrice) to avoid double counting.
   403→  // When the pre-order is validated, a Purchase entry is created with the pre-order total
   404→  // (which includes this item's cost). If we also set purchaseCost > 0 on the StockItem,
   405→  // the accounting API would count it TWICE in the ACHATS register:
   406→  //   1. Once via StockItem.purchaseCost (in the stock-items-with-purchaseDate-in-period query)
   407→  //   2. Once via Purchase.amount (the pre-order total)
   408→  // By setting purchaseCost = 0, only the Purchase entry counts — which is correct.
   409→  const createStockItem = async (idx: number) => {
   410→    const item = items[idx]
   411→    if (!item.designation.trim()) {
   412→      toast.error('Veuillez saisir une désignation avant de créer l\'article')
   413→      return
   414→    }
   415→    setCreatingArticleIdx(idx)
   416→    try {
   417→      // Generate a unique SKU
   418→      const sku = `ART-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).substring(2, 5).toUpperCase()}`
   419→      const res = await fetch('/api/stock', {
   420→        method: 'POST',
   421→        headers: { 'Content-Type': 'application/json' },
   422→        body: JSON.stringify({
   423→          sku,
   424→          title: item.designation.trim(),
   425→          brand: item.designation.trim().split(' ')[0] || 'Article',  // first word as brand fallback
   426→          category: categories[0]?.code || 'vetements',
   427→          size: item.size || null,
   428→          color: item.color || null,
   429→          condition: item.condition || conditions[0]?.code || 'bon',
   430→          purchaseCost: 0,  // 0 to avoid double counting (pre-order Purchase already accounts for the cost)
   431→          purchaseDate: orderDate,
   432→          supplierId: supplierId || null,
   433→          quantity: 0,  // quantité = 0 comme demandé
   434→          description: item.description || null,
   435→          status: 'A_PHOTOGRAPHIER',
   436→        }),
   437→      })
   438→      const data = await res.json()
   439→      if (!res.ok) { toast.error(data.error || 'Erreur lors de la création'); return }
   440→      // Link the created StockItem back to the pre-order item
   441→      updateItem(idx, 'stockItemId', data.id)
   442→      toast.success(`Article créé dans le stock (SKU: ${sku}, qté: 0, coût: 0€ — comptabilisé via la pré-commande)`)
   443→    } catch {
   444→      toast.error('Erreur réseau')
   445→    } finally {
   446→      setCreatingArticleIdx(null)
   447→    }
   448→  }
   449→
   450→  return (
   451→    <div className="space-y-4">
   452→      <div className="flex items-center gap-3">
   453→        <Button variant="ghost" size="sm" onClick={onBack}>
   454→          <ArrowLeft className="h-4 w-4 mr-1" /> Retour
   455→        </Button>
   456→        <h1 className="text-2xl font-bold">Nouvelle pré-commande</h1>
   457→      </div>
   458→
   459→      {/* Infos générales */}
   460→      <Card>
   461→        <CardHeader>
   462→          <CardTitle className="text-base">Informations générales</CardTitle>
   463→        </CardHeader>
   464→        <CardContent className="grid grid-cols-1 md:grid-cols-3 gap-4">
   465→          <div className="space-y-1.5 md:col-span-2">
   466→            <Label>Nom de la pré-commande *</Label>
   467→            <Input value={name} onChange={e => setName(e.target.value)} placeholder="Ex: Achat rentrée septembre" />
   468→          </div>
   469→          <div className="space-y-1.5">
   470→            <Label>Date</Label>
   471→            <Input type="date" value={orderDate} onChange={e => setOrderDate(e.target.value)} />
   472→          </div>
   473→          <div className="space-y-1.5 md:col-span-3">
   474→            <Label>Fournisseur</Label>
   475→            <Select value={supplierId} onValueChange={setSupplierId}>
   476→              <SelectTrigger><SelectValue placeholder="Sélectionner un fournisseur (optionnel)" /></SelectTrigger>
   477→              <SelectContent>
   478→                {(suppliers || []).map(s => (
   479→                  <SelectItem key={s.id} value={s.id}>{s.name} ({s.type})</SelectItem>
   480→                ))}
   481→              </SelectContent>
   482→            </Select>
   483→          </div>
   484→          <div className="space-y-1.5 md:col-span-3">
   485→            <Label>Méthode de paiement</Label>
   486→            <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   487→              <SelectTrigger><SelectValue placeholder="Sélectionner un mode de paiement (optionnel)" /></SelectTrigger>
   488→              <SelectContent>
   489→                <SelectItem value="especes">Espèces</SelectItem>
   490→                <SelectItem value="carte_bancaire">Carte bancaire</SelectItem>
   491→                <SelectItem value="virement">Virement</SelectItem>
   492→                <SelectItem value="cheque">Chèque</SelectItem>
   493→                <SelectItem value="paypal">PayPal</SelectItem>
   494→              </SelectContent>
   495→            </Select>
   496→          </div>
   497→        </CardContent>

   691→  const isAdmin = session?.user?.role === 'admin'
   692→  const { data: preorder, loading, refresh } = useFetch<PreOrder>(`/api/preorders/${id}`)
   693→  const [saving, setSaving] = useState(false)
   694→  const [validating, setValidating] = useState(false)
   695→  const [deleting, setDeleting] = useState(false)

   703→  const items = preorder ? parseItems(preorder.items) : []
   704→  const cfg = preorder ? (STATUS_CONFIG[preorder.status] || STATUS_CONFIG.pending) : null
   705→  const isValidated = preorder?.status === 'validated'
   706→  const isCancelled = preorder?.status === 'cancelled'

   708→  const isReceived = preorder?.status === 'received'

   712→    try {
   713→      const res = await fetch(`/api/preorders/${id}`, {
   714→        method: 'PATCH',
   715→        headers: { 'Content-Type': 'application/json' },
   716→        body: JSON.stringify(patch),
   717→      })
   718→      const data = await res.json()
   719→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }
   720→      toast.success('Enregistré')
   721→      refresh()
   722→    } catch {
   723→      toast.error('Erreur réseau')

   726→    }

   774→    } catch {
   775→      toast.error('Erreur réseau')
   776→    } finally {
   777→      setValidating(false)
   778→    }

   783→      <div className="flex justify-center py-12">
   784→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   785→      </div>

   809→          {isPending && (

  1110→// ── Product picker dialog ────────────────────────────────────────────────
  1111→// Modal with category → subcategory filters, product grid (photo + name), and pagination.
  1112→
  1113→interface ProductPickerProps {
  1114→  open: boolean
  1115→  onOpenChange: (o: boolean) => void
  1116→  stockItems: StockItemLite[]
  1117→  categories: { id: string; code: string; value: string }[]
  1118→  getSubcategories: (parentCode: string | null | undefined) => { id: string; code: string; value: string; parentCode: string | null }[]
  1119→  onPick: (stockItemId: string) => void
  1120→}
  1121→
  1122→const PAGE_SIZE = 12
  1123→
  1124→function ProductPickerDialog({ open, onOpenChange, stockItems, categories, getSubcategories, onPick }: ProductPickerProps) {
  1125→  const [categoryFilter, setCategoryFilter] = useState<string>('')
  1126→  const [subcategoryFilter, setSubcategoryFilter] = useState<string>('')
  1127→  const [search, setSearch] = useState('')
  1128→  const [page, setPage] = useState(1)
  1129→
  1130→  // Reset filters when dialog opens
  1131→  useEffect(() => {
  1132→    if (open) {
  1133→      setCategoryFilter('')
  1134→      setSubcategoryFilter('')
  1135→      setSearch('')
  1136→      setPage(1)
  1137→    }
  1138→  }, [open])
  1139→
  1140→  const subcategories = categoryFilter ? getSubcategories(categoryFilter) : []
  1141→
  1142→  // Filter stock items
  1143→  const filtered = useMemo(() => {
  1144→    let list = stockItems
  1145→    if (categoryFilter) list = list.filter(si => si.category === categoryFilter)
  1146→    if (subcategoryFilter) list = list.filter(si => si.subcategory === subcategoryFilter)
  1147→    if (search) {
  1148→      const q = search.toLowerCase()
  1149→      list = list.filter(si =>
  1150→        (si.title || '').toLowerCase().includes(q) ||
  1151→        si.brand.toLowerCase().includes(q) ||
  1152→        si.sku.toLowerCase().includes(q)
  1153→      )
  1154→    }
  1155→    return list
  1156→  }, [stockItems, categoryFilter, subcategoryFilter, search])
  1157→
  1158→  // Pagination
  1159→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE))
  1160→  const currentPage = Math.min(page, totalPages)
  1161→  const paginated = filtered.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE)
  1162→
  1163→  // Helper to parse first photo
  1164→  const getPhoto = (photosJson: string): string | null => {
  1165→    try {
  1166→      const arr = JSON.parse(photosJson)
  1167→      if (Array.isArray(arr) && arr.length > 0) {
  1168→        const p = arr[0]
  1169→        return p.startsWith('/uploads/') ? `/api${p}` : p
  1170→      }
  1171→    } catch {}
  1172→    return null
  1173→  }
  1174→
  1175→  return (
  1176→    <Dialog open={open} onOpenChange={onOpenChange}>
  1177→      <DialogContent className="max-w-4xl max-h-[85vh] overflow-hidden flex flex-col">
  1178→        <DialogHeader>
  1179→          <DialogTitle>Rechercher un article existant</DialogTitle>
  1180→          <DialogDescription>
  1181→            Filtrez par catégorie et sous-catégorie, puis cliquez sur un article pour l'ajouter à la pré-commande.
  1182→          </DialogDescription>
  1183→        </DialogHeader>
  1184→
  1185→        {/* Filters */}
  1186→        <div className="grid grid-cols-1 md:grid-cols-3 gap-3 pb-3 border-b">
  1187→          <div className="space-y-1.5">
  1188→            <Label className="text-xs">Catégorie</Label>
  1189→            <Select
  1190→              value={categoryFilter || '__all__'}
  1191→              onValueChange={(v) => { setCategoryFilter(v === '__all__' ? '' : v); setSubcategoryFilter(''); setPage(1) }}
  1192→            >
  1193→              <SelectTrigger><SelectValue placeholder="Toutes" /></SelectTrigger>
  1194→              <SelectContent>
  1195→                <SelectItem value="__all__">Toutes les catégories</SelectItem>
  1196→                {categories.map(c => <SelectItem key={c.id} value={c.code}>{c.value}</SelectItem>)}
  1197→              </SelectContent>
  1198→            </Select>
  1199→          </div>
  1200→          <div className="space-y-1.5">
  1201→            <Label className="text-xs">Sous-catégorie</Label>
  1202→            <Select
  1203→              value={subcategoryFilter || '__all__'}
  1204→              onValueChange={(v) => { setSubcategoryFilter(v === '__all__' ? '' : v); setPage(1) }}
  1205→              disabled={!categoryFilter || subcategories.length === 0}
  1206→            >
  1207→              <SelectTrigger><SelectValue placeholder="Toutes" /></SelectTrigger>
  1208→              <SelectContent>
  1209→                <SelectItem value="__all__">Toutes les sous-catégories</SelectItem>
  1210→                {subcategories.map(s => <SelectItem key={s.id} value={s.code}>{s.value}</SelectItem>)}
  1211→              </SelectContent>
  1212→            </Select>
  1213→          </div>
  1214→          <div className="space-y-1.5">
  1215→            <Label className="text-xs">Recherche</Label>

  1221→          </div>
  1222→        </div>
  1223→
  1224→        {/* Results count */}

  1228→
  1229→        {/* Product grid */}
  1230→        <div className="flex-1 overflow-y-auto">
