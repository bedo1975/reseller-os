    86→function fmtDate(iso: string): string {
    87→  try { return new Date(iso).toLocaleDateString('fr-FR') } catch { return '—' }

    91→  return (n || 0).toFixed(2) + ' €'

   299→function CreatePreOrderForm({ onBack, onCreated }: { onBack: () => void; onCreated: () => void }) {
   300→  const { data: suppliers } = useFetch<Supplier[]>('/api/suppliers')
   301→  const { data: stockItems } = useFetch<StockItemLite[]>('/api/stock')
   302→  const { getByType } = useSettings()
   303→  const sizes = getByType('size')
   304→  const colors = getByType('color')
   305→  const conditions = getByType('condition')
   306→  const categories = getByType('category')
   307→  const [saving, setSaving] = useState(false)
   308→  const [creatingArticleIdx, setCreatingArticleIdx] = useState<number | null>(null)
   309→
   310→  const [name, setName] = useState('')
   311→  const [supplierId, setSupplierId] = useState<string>('')
   312→  const [orderDate, setOrderDate] = useState(new Date().toISOString().slice(0, 10))
   313→  const [paymentMethod, setPaymentMethod] = useState<string>('')
   314→  const [notes, setNotes] = useState('')
   315→  const [shippingCost, setShippingCost] = useState(0)
   316→  const [items, setItems] = useState<PreOrderItem[]>([
   317→    { designation: '', url: '', description: '', size: '', color: '', condition: '', quantity: 1, unitPrice: 0, stockItemId: null },
   318→  ])
   319→
   320→  const subtotal = useMemo(() => {
   321→    return items.reduce((s, it) => s + (Number(it.quantity) || 0) * (Number(it.unitPrice) || 0), 0)
   322→  }, [items])
   323→  const total = subtotal + (Number(shippingCost) || 0)
   324→
   325→  const updateItem = (idx: number, field: keyof PreOrderItem, value: any) => {
   326→    setItems(prev => prev.map((it, i) => i === idx ? { ...it, [field]: value } : it))
   327→  }
   328→
   329→  const addItem = () => {
   330→    setItems(prev => [...prev, { designation: '', url: '', description: '', size: '', color: '', condition: '', quantity: 1, unitPrice: 0, stockItemId: null }])
   331→  }
   332→
   333→  const removeItem = (idx: number) => {
   334→    setItems(prev => prev.length > 1 ? prev.filter((_, i) => i !== idx) : prev)
   335→  }
   336→
   337→  // When selecting an existing stock item, auto-fill the fields
   338→  const selectStockItem = (idx: number, stockItemId: string) => {
   339→    if (!stockItemId) {
   340→      updateItem(idx, 'stockItemId', null)
   341→      return
   342→    }
   343→    const item = (stockItems || []).find(i => i.id === stockItemId)
   344→    if (!item) return
   345→    setItems(prev => prev.map((it, i) => {
   346→      if (i !== idx) return it
   347→      const designation = item.title || `${item.brand} ${item.category}`.trim()
   348→      return {
   349→        ...it,
   350→        stockItemId: item.id,
   351→        designation,
   352→        size: item.size || '',
   353→        color: item.color || '',
   354→        condition: item.condition || '',
   355→      }
   356→    }))
   357→  }
   358→
   359→  const submit = async () => {
   360→    if (!name.trim()) { toast.error('Le nom est requis'); return }
   361→    if (items.length === 0 || items.every(i => !i.designation.trim())) {
   362→      toast.error('Au moins un article avec désignation est requis'); return
   363→    }
   364→    // Filter out empty items
   365→    const cleanItems = items.filter(i => i.designation.trim())
   366→    if (cleanItems.length === 0) { toast.error('Au moins un article est requis'); return }
   367→
   368→    setSaving(true)
   369→    try {
   370→      const res = await fetch('/api/preorders', {
   371→        method: 'POST',
   372→        headers: { 'Content-Type': 'application/json' },
   373→        body: JSON.stringify({
   374→          name: name.trim(),
   375→          supplierId: supplierId || null,
   376→          supplierName: supplierId ? (suppliers || []).find(s => s.id === supplierId)?.name : null,
   377→          orderDate,
   378→          paymentMethod: paymentMethod || null,
   379→          items: cleanItems,
   380→          shippingCost: Number(shippingCost) || 0,
   381→          notes: notes.trim() || null,
   382→        }),
   383→      })
   384→      const data = await res.json()
   385→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }
   386→      toast.success('Pré-commande créée')
   387→      onCreated()
   388→    } catch {
   389→      toast.error('Erreur réseau')
   390→    } finally {
   391→      setSaving(false)
   392→    }
   393→  }
   394→
   395→  // Create a StockItem from the current article line (qty = 0, status = A_PHOTOGRAPHIER)
   396→  // and link it back to the pre-order item.
   397→  //
   398→  // IMPORTANT: purchaseCost is set to 0 (not item.unitPrice) to avoid double counting.
   399→  // When the pre-order is validated, a Purchase entry is created with the pre-order total
   400→  // (which includes this item's cost). If we also set purchaseCost > 0 on the StockItem,
   401→  // the accounting API would count it TWICE in the ACHATS register:
   402→  //   1. Once via StockItem.purchaseCost (in the stock-items-with-purchaseDate-in-period query)
   403→  //   2. Once via Purchase.amount (the pre-order total)
   404→  // By setting purchaseCost = 0, only the Purchase entry counts — which is correct.
   405→  const createStockItem = async (idx: number) => {
   406→    const item = items[idx]
   407→    if (!item.designation.trim()) {
   408→      toast.error('Veuillez saisir une désignation avant de créer l\'article')
   409→      return
   410→    }
   411→    setCreatingArticleIdx(idx)
   412→    try {
   413→      // Generate a unique SKU
   414→      const sku = `ART-${Date.now().toString(36).toUpperCase()}-${Math.random().toString(36).substring(2, 5).toUpperCase()}`
   415→      const res = await fetch('/api/stock', {
   416→        method: 'POST',
   417→        headers: { 'Content-Type': 'application/json' },
   418→        body: JSON.stringify({
   419→          sku,
   420→          title: item.designation.trim(),
   421→          brand: item.designation.trim().split(' ')[0] || 'Article',  // first word as brand fallback
   422→          category: categories[0]?.code || 'vetements',
   423→          size: item.size || null,
   424→          color: item.color || null,
   425→          condition: item.condition || conditions[0]?.code || 'bon',
   426→          purchaseCost: 0,  // 0 to avoid double counting (pre-order Purchase already accounts for the cost)
   427→          purchaseDate: orderDate,
   428→          supplierId: supplierId || null,
   429→          quantity: 0,  // quantité = 0 comme demandé
   430→          description: item.description || null,
   431→          status: 'A_PHOTOGRAPHIER',
   432→        }),
   433→      })
   434→      const data = await res.json()
   435→      if (!res.ok) { toast.error(data.error || 'Erreur lors de la création'); return }
   436→      // Link the created StockItem back to the pre-order item
   437→      updateItem(idx, 'stockItemId', data.id)
   438→      toast.success(`Article créé dans le stock (SKU: ${sku}, qté: 0, coût: 0€ — comptabilisé via la pré-commande)`)
   439→    } catch {
   440→      toast.error('Erreur réseau')
   441→    } finally {
   442→      setCreatingArticleIdx(null)
   443→    }
   444→  }
   445→
   446→  return (
   447→    <div className="space-y-4">
   448→      <div className="flex items-center gap-3">
   449→        <Button variant="ghost" size="sm" onClick={onBack}>
   450→          <ArrowLeft className="h-4 w-4 mr-1" /> Retour
   451→        </Button>
   452→        <h1 className="text-2xl font-bold">Nouvelle pré-commande</h1>
   453→      </div>
   454→
   455→      {/* Infos générales */}
   456→      <Card>
   457→        <CardHeader>
   458→          <CardTitle className="text-base">Informations générales</CardTitle>
   459→        </CardHeader>
   460→        <CardContent className="grid grid-cols-1 md:grid-cols-3 gap-4">
   461→          <div className="space-y-1.5 md:col-span-2">
   462→            <Label>Nom de la pré-commande *</Label>
   463→            <Input value={name} onChange={e => setName(e.target.value)} placeholder="Ex: Achat rentrée septembre" />
   464→          </div>
   465→          <div className="space-y-1.5">
   466→            <Label>Date</Label>
   467→            <Input type="date" value={orderDate} onChange={e => setOrderDate(e.target.value)} />
   468→          </div>
   469→          <div className="space-y-1.5 md:col-span-3">
   470→            <Label>Fournisseur</Label>
   471→            <Select value={supplierId} onValueChange={setSupplierId}>
   472→              <SelectTrigger><SelectValue placeholder="Sélectionner un fournisseur (optionnel)" /></SelectTrigger>
   473→              <SelectContent>
   474→                {(suppliers || []).map(s => (
   475→                  <SelectItem key={s.id} value={s.id}>{s.name} ({s.type})</SelectItem>
   476→                ))}
   477→              </SelectContent>
   478→            </Select>
   479→          </div>
   480→          <div className="space-y-1.5 md:col-span-3">
   481→            <Label>Méthode de paiement</Label>
   482→            <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   483→              <SelectTrigger><SelectValue placeholder="Sélectionner un mode de paiement (optionnel)" /></SelectTrigger>
   484→              <SelectContent>
   485→                <SelectItem value="especes">Espèces</SelectItem>
   486→                <SelectItem value="carte_bancaire">Carte bancaire</SelectItem>
   487→                <SelectItem value="virement">Virement</SelectItem>
   488→                <SelectItem value="cheque">Chèque</SelectItem>
   489→                <SelectItem value="paypal">PayPal</SelectItem>
   490→              </SelectContent>
   491→            </Select>
   492→          </div>
   493→        </CardContent>
   494→      </Card>
   495→
   496→      {/* Articles */}
   497→      <Card>
   498→        <CardHeader>
   499→          <div className="flex items-center justify-between">

   655→function PreOrderDetail({ id, onBack }: { id: string; onBack: () => void }) {
   656→  const { data: session } = useSession()
   657→  const isAdmin = session?.user?.role === 'admin'
   658→  const { data: preorder, loading, refresh } = useFetch<PreOrder>(`/api/preorders/${id}`)
   659→  const [saving, setSaving] = useState(false)
   660→  const [validating, setValidating] = useState(false)
   661→  const [deleting, setDeleting] = useState(false)
   662→  const [receiving, setReceiving] = useState(false)
   663→  const [showValidateDialog, setShowValidateDialog] = useState(false)
   664→  const [showDeleteDialog, setShowDeleteDialog] = useState(false)
   665→  const [showReceiveDialog, setShowReceiveDialog] = useState(false)
   666→  const [orderNumber, setOrderNumber] = useState('')
   667→  const [invoiceNumber, setInvoiceNumber] = useState('')
   668→
   669→  const items = preorder ? parseItems(preorder.items) : []
   670→  const cfg = preorder ? (STATUS_CONFIG[preorder.status] || STATUS_CONFIG.pending) : null
   671→  const isValidated = preorder?.status === 'validated'
   672→  const isCancelled = preorder?.status === 'cancelled'
   673→  const isPending = preorder?.status === 'pending'
   674→  const isReceived = preorder?.status === 'received'
   675→
   676→  const save = async (patch: any) => {
   677→    setSaving(true)
   678→    try {
   679→      const res = await fetch(`/api/preorders/${id}`, {
   680→        method: 'PATCH',
   681→        headers: { 'Content-Type': 'application/json' },
   682→        body: JSON.stringify(patch),
   683→      })
   684→      const data = await res.json()
   685→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }
   686→      toast.success('Enregistré')
   687→      refresh()
   688→    } catch {
   689→      toast.error('Erreur réseau')
   690→    } finally {
   691→      setSaving(false)
   692→    }
   693→  }
   694→
   695→  const deletePreorder = async () => {
   696→    setDeleting(true)
   697→    try {
   698→      const res = await fetch(`/api/preorders/${id}`, { method: 'DELETE' })

   700→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }

   716→      if (!res.ok) { toast.error(data.error || 'Erreur'); return }
   717→      toast.success(data.message || 'Commande reçue — articles ajoutés au stock')
   718→      setShowReceiveDialog(false)
   719→      refresh()
   720→    } catch {
   721→      toast.error('Erreur réseau')
   722→    } finally {
   723→      setReceiving(false)
   724→    }
   725→  }
   726→
   727→  const validate = async () => {
   728→    setValidating(true)
   729→    try {
   730→      const res = await fetch(`/api/preorders/${id}/validate`, {
   731→        method: 'POST',
   732→        headers: { 'Content-Type': 'application/json' },
   733→        body: JSON.stringify({ orderNumber, invoiceNumber }),
   734→      })

   738→      setShowValidateDialog(false)
   739→      refresh()
   740→    } catch {
   741→      toast.error('Erreur réseau')
   742→    } finally {
   743→      setValidating(false)
   744→    }
   745→  }

   750→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
