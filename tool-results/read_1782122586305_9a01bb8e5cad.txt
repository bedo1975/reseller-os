    31→export function TaxesModule() {
    32→  const [activeTab, setActiveTab] = useState<Tab>('synthese')
    33→  const [year, setYear] = useState(new Date().getFullYear())
    34→
    35→  const tabs = [
    36→    { id: 'synthese' as Tab, label: 'Synthèse', icon: BarChart3, description: 'Vue d\'ensemble + dépenses' },
    37→    { id: 'recettes' as Tab, label: 'Livre des recettes', icon: BookOpen, description: 'Recettes chronologiques (légal)' },
    38→    { id: 'achats' as Tab, label: 'Registre des achats', icon: ClipboardList, description: 'Achats par article (légal)' },
    39→  ]
    40→
    41→  return (
    42→    <div className="space-y-5">
    43→      {/* Tabs */}
    44→      <div className="flex flex-wrap gap-2">
    45→        {tabs.map(t => {
    46→          const Icon = t.icon
    47→          const active = activeTab === t.id
    48→          return (
    49→            <button
    50→              key={t.id}
    51→              onClick={() => setActiveTab(t.id)}
    52→              className={cn(
    53→                'px-4 py-2 rounded-lg text-sm font-medium transition-all border flex items-center gap-2',
    54→                active
    55→                  ? 'bg-foreground text-background border-foreground'
    56→                  : 'bg-card hover:bg-muted border-border'
    57→              )}
    58→            >
    59→              <Icon className="h-4 w-4" />
    60→              <span>{t.label}</span>
    61→            </button>
    62→          )
    63→        })}
    64→      </div>
    65→
    66→      {/* Year selector (commun) */}
    67→      <Card>
    68→        <CardContent className="p-4 flex items-center justify-between">
    69→          <div className="flex items-center gap-2">
    70→            <Receipt className="h-4 w-4 text-amber-600" />
    71→            <span className="text-sm font-medium">Année fiscale :</span>
    72→            <Select value={String(year)} onValueChange={v => setYear(parseInt(v))}>
    73→              <SelectTrigger className="w-[110px] h-8">
    74→                <SelectValue />
    75→              </SelectTrigger>
    76→              <SelectContent>
    77→                {[2024, 2025, 2026, 2027].map(y => (
    78→                  <SelectItem key={y} value={String(y)}>{y}</SelectItem>
    79→                ))}
    80→              </SelectContent>
    81→            </Select>
    82→          </div>
    83→          <p className="text-xs text-muted-foreground hidden sm:block">
    84→            {tabs.find(t => t.id === activeTab)?.description}
    85→          </p>
    86→        </CardContent>
    87→      </Card>
    88→
    89→      {activeTab === 'synthese' && <SyntheseTab year={year} />}
    90→      {activeTab === 'recettes' && <RecettesTab year={year} />}
    91→      {activeTab === 'achats' && <AchatsTab year={year} />}
    92→    </div>
    93→  )
    94→}

    97→// ONGLET 1 — SYNTHÈSE (le module Fiscalité existant)

    99→
   100→function SyntheseTab({ year }: { year: number }) {
   101→  const { data: sales, loading: salesLoading } = useFetch<Sale[]>('/api/sales')
   102→  const { data: expenses, loading: expLoading, refresh } = useFetch<Expense[]>('/api/expenses')
   103→  const [showForm, setShowForm] = useState(false)
   104→  const [form, setForm] = useState({
   105→    date: new Date().toISOString().split('T')[0],
   106→    category: 'frais_port',
   107→    label: '',
   108→    amount: '',
   109→  })
   110→  const [saving, setSaving] = useState(false)
   111→
   112→  const yearSales = useMemo(() =>
   113→    (sales || []).filter(s => new Date(s.saleDate).getFullYear() === year),
   114→    [sales, year]
   115→  )
   116→
   117→  const yearExpenses = useMemo(() =>
   118→    (expenses || []).filter(e => new Date(e.date).getFullYear() === year),
   119→    [expenses, year]
   120→  )
   121→
   122→  const totalCA = yearSales.reduce((s, x) => s + x.salePrice, 0)
   123→  const totalPurchases = yearSales.reduce((s, x) => s + x.stockItem.purchaseCost, 0)
   124→  const totalPlatformFees = yearSales.reduce((s, x) => s + (x.platformFees || 0) + (x.platformFixedFees || 0), 0)
   125→  const totalShipping = yearSales.reduce((s, x) => s + x.shippingCost, 0)
   126→  const totalOtherExpenses = yearExpenses.reduce((s, e) => s + e.amount, 0)
   127→  const totalProfit = yearSales.reduce((s, x) => s + x.profit, 0) - totalOtherExpenses
   128→
   129→  const exportCSV = () => {
   130→    const rows: string[][] = []
   131→    rows.push(['Type', 'Date', 'Plateforme', 'Article', 'Description', 'CA', 'Coût', 'Frais plateforme', 'Frais port', 'Profit', 'Marge %'])
   132→    yearSales.forEach(s => {
   133→      rows.push([
   134→        'Vente', formatDate(s.saleDate), s.platform, s.stockItem.sku,
   135→        `${s.stockItem.brand} ${s.stockItem.size || ''} ${s.stockItem.color || ''}`.trim(),
   136→        s.salePrice.toFixed(2), s.stockItem.purchaseCost.toFixed(2),
   137→        ((s.platformFees || 0) + (s.platformFixedFees || 0)).toFixed(2),
   138→        s.shippingCost.toFixed(2), s.profit.toFixed(2), s.margin.toString(),
   139→      ])
   140→    })
   141→    yearExpenses.forEach(e => {
   142→      rows.push(['Dépense', formatDate(e.date), '', '', e.label, '', '', '', '', (-e.amount).toFixed(2), ''])
   143→    })
   144→    rows.push([])
   145→    rows.push(['TOTAUX', '', '', '', '', totalCA.toFixed(2), totalPurchases.toFixed(2), totalPlatformFees.toFixed(2), totalShipping.toFixed(2), totalProfit.toFixed(2), ''])
   146→
   147→    const csv = rows.map(r => r.map(c => `"${(c || '').replace(/"/g, '""')}"`).join(',')).join('\n')
   148→    const blob = new Blob(['\ufeff' + csv], { type: 'text/csv;charset=utf-8' })
   149→    const url = URL.createObjectURL(blob)
   150→    const a = document.createElement('a')
   151→    a.href = url
   152→    a.download = `reseller-os-fiscalite-${year}.csv`
   153→    a.click()
   154→    URL.revokeObjectURL(url)
   155→    toast.success('Export CSV téléchargé')
   156→  }
   157→
   158→  const exportExcel = () => {
   159→    const rows: string[][] = []
   160→    rows.push(['Type', 'Date', 'Plateforme', 'SKU', 'Description', 'CA (€)', 'Coût (€)', 'Frais plateforme (€)', 'Frais port (€)', 'Profit (€)', 'Marge (%)'])
   161→    yearSales.forEach(s => {
   162→      rows.push([
   163→        'Vente', formatDate(s.saleDate), s.platform, s.stockItem.sku,
   164→        `${s.stockItem.brand} ${s.stockItem.size || ''} ${s.stockItem.color || ''}`.trim(),
   165→        s.salePrice.toFixed(2), s.stockItem.purchaseCost.toFixed(2),
   166→        ((s.platformFees || 0) + (s.platformFixedFees || 0)).toFixed(2),
   167→        s.shippingCost.toFixed(2), s.profit.toFixed(2), s.margin.toString(),
   168→      ])
   169→    })
   170→    yearExpenses.forEach(e => {
   171→      rows.push(['Dépense', formatDate(e.date), '', '', e.label, '', '', '', '', (-e.amount).toFixed(2), ''])
   172→    })
   173→    const tsv = rows.map(r => r.join('\t')).join('\n')
   174→    const blob = new Blob(['\ufeff' + tsv], { type: 'application/vnd.ms-excel;charset=utf-8' })
   175→    const url = URL.createObjectURL(blob)
   176→    const a = document.createElement('a')
   177→    a.href = url
   178→    a.download = `reseller-os-fiscalite-${year}.xls`
   179→    a.click()
   180→    URL.revokeObjectURL(url)
   181→    toast.success('Export Excel téléchargé')
   182→  }
   183→
   184→  const exportPDF = () => {
   185→    const win = window.open('', '_blank')
   186→    if (!win) {
   187→      toast.error('Bloqueur de pop-up : autorisez les pop-ups pour exporter en PDF')
   188→      return
   189→    }
   190→    win.document.write(`
   191→      <!DOCTYPE html>
   192→      <html lang="fr"><head><meta charset="utf-8">
   193→      <title>Reseller OS — Fiscalité ${year}</title>
   194→      <style>
   195→        * { box-sizing: border-box; }
   196→        body { font-family: -apple-system, 'Segoe UI', Roboto, sans-serif; margin: 40px; color: #1a1a1a; }
   197→        h1 { font-size: 24px; margin: 0 0 4px; }
   198→        .subtitle { color: #666; margin-bottom: 24px; font-size: 13px; }
   199→        .summary { display: grid; grid-template-columns: repeat(4, 1fr); gap: 12px; margin-bottom: 28px; }
   200→        .summary div { background: #f7f7f7; padding: 14px; border-radius: 8px; }
   201→        .summary .label { font-size: 11px; color: #666; text-transform: uppercase; }
   202→        .summary .value { font-size: 18px; font-weight: 700; margin-top: 4px; }
   203→        table { width: 100%; border-collapse: collapse; font-size: 12px; margin-bottom: 28px; }
   204→        th, td { padding: 8px 10px; text-align: left; border-bottom: 1px solid #eee; }
   205→        th { background: #f7f7f7; font-weight: 600; text-transform: uppercase; font-size: 10px; color: #666; }
   206→        .total { font-weight: 700; background: #f0fdf4 !important; }
   207→        .profit { color: #10b981; font-weight: 600; }
   208→        h2 { font-size: 16px; margin-top: 24px; }
   209→      </style></head>
   210→      <body>
   211→        <h1>Reseller OS — Rapport fiscal ${year}</h1>
   212→        <p class="subtitle">Généré le ${new Date().toLocaleDateString('fr-FR')}</p>
   213→        <div class="summary">
   214→          <div><div class="label">CA</div><div class="value">${formatEUR(totalCA)}</div></div>
   215→          <div><div class="label">Achats</div><div class="value">${formatEUR(totalPurchases)}</div></div>
   216→          <div><div class="label">Frais plateforme</div><div class="value">${formatEUR(totalPlatformFees)}</div></div>
   217→          <div><div class="label">Frais port</div><div class="value">${formatEUR(totalShipping)}</div></div>
   218→          <div><div class="label">Autres dépenses</div><div class="value">${formatEUR(totalOtherExpenses)}</div></div>
   219→          <div><div class="label">Bénéfice net</div><div class="value profit">${formatEUR(totalProfit)}</div></div>
   220→          <div><div class="label">Marge</div><div class="value">${totalCA > 0 ? ((totalProfit / totalCA) * 100).toFixed(1) : 0}%</div></div>
   221→          <div><div class="label">Nb ventes</div><div class="value">${yearSales.length}</div></div>
   222→        </div>
   223→        <h2>Ventes</h2>
   224→        <table><thead><tr><th>Date</th><th>Plateforme</th><th>Article</th><th>CA</th><th>Coût</th><th>Frais</th><th>Profit</th></tr></thead>
   225→        <tbody>
   226→          ${yearSales.map(s => `
   227→            <tr>
   228→              <td>${formatDate(s.saleDate)}</td>
   229→              <td>${s.platform}</td>
   230→              <td>${s.stockItem.brand} (${s.stockItem.sku})</td>
   231→              <td>${formatEUR(s.salePrice)}</td>
   232→              <td>${formatEUR(s.stockItem.purchaseCost)}</td>
   233→              <td>${formatEUR((s.platformFees || 0) + (s.shippingCost || 0) + (s.platformFixedFees || 0))}</td>
   234→              <td class="profit">${formatEUR(s.profit)}</td>
   235→            </tr>
   236→          `).join('')}
   237→          <tr class="total">
   238→            <td colspan="3">TOTAUX</td>
   239→            <td>${formatEUR(totalCA)}</td>
   240→            <td>${formatEUR(totalPurchases)}</td>
   241→            <td>${formatEUR(totalPlatformFees + totalShipping)}</td>
   242→            <td class="profit">${formatEUR(yearSales.reduce((s, x) => s + x.profit, 0))}</td>
   243→          </tr>
   244→        </tbody></table>
   245→        ${yearExpenses.length > 0 ? `
   246→          <h2>Autres dépenses</h2>
   247→          <table><thead><tr><th>Date</th><th>Catégorie</th><th>Libellé</th><th>Montant</th></tr></thead>
   248→          <tbody>
   249→            ${yearExpenses.map(e => `
   250→              <tr><td>${formatDate(e.date)}</td><td>${e.category}</td><td>${e.label}</td><td>${formatEUR(e.amount)}</td></tr>
   251→            `).join('')}
   252→            <tr class="total"><td colspan="3">TOTAL DÉPENSES</td><td>${formatEUR(totalOtherExpenses)}</td></tr>
   253→          </tbody></table>
   254→        ` : ''}
   255→        <script>window.onload = () => { window.print(); }</script>
   256→      </body></html>
   257→    `)
   258→    win.document.close()
   259→    toast.success('Export PDF ouvert — utilisez Ctrl/Cmd+P pour enregistrer')
   260→  }
   261→
   262→  const addExpense = async () => {
   263→    if (!form.label || !form.amount) {
   264→      toast.error('Libellé et montant requis')
   265→      return
   266→    }
   267→    setSaving(true)
   268→    try {
   269→      const res = await fetch('/api/expenses', {
   270→        method: 'POST',
   271→        headers: { 'Content-Type': 'application/json' },
   272→        body: JSON.stringify(form),
   273→      })
   274→      if (!res.ok) throw new Error('Erreur')
   275→      toast.success('Dépense ajoutée')
   276→      setForm({ date: new Date().toISOString().split('T')[0], category: 'frais_port', label: '', amount: '' })
   277→      setShowForm(false)
   278→      refresh()
   279→    } catch {
   280→      toast.error('Erreur')
   281→    } finally {
   282→      setSaving(false)
   283→    }
   284→  }
   285→
   286→  const deleteExpense = async (id: string) => {
   287→    if (!confirm('Supprimer cette dépense ?')) return
   288→    await fetch(`/api/expenses/${id}`, { method: 'DELETE' })
   289→    toast.success('Dépense supprimée')
   290→    refresh()
   291→  }
   292→
   293→  const loading = salesLoading || expLoading
   294→
   295→  return (
   296→    <div className="space-y-5">
   297→      {/* Hero avec exports */}
   298→      <Card className="bg-gradient-to-br from-amber-50 via-white to-amber-50/30 dark:from-amber-950/30 dark:via-card dark:to-amber-950/20 border-amber-200 dark:border-amber-900">
   299→        <CardContent className="p-6">
   300→          <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4">
   301→            <div>
   302→              <div className="flex items-center gap-2 mb-2">
   303→                <Receipt className="h-5 w-5 text-amber-600" />
   304→                <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 hover:bg-amber-100">
   305→                  Synthèse fiscale {year}
   306→                </Badge>
   307→              </div>
   308→              <p className="text-3xl font-bold tracking-tight">{formatEUR(totalProfit)} <span className="text-base font-normal text-muted-foreground">bénéfice net</span></p>
   309→              <p className="text-sm text-muted-foreground mt-1">
   310→                CA {formatEUR(totalCA)} · Marge nette {totalCA > 0 ? ((totalProfit / totalCA) * 100).toFixed(1) : 0}%
   311→              </p>
   312→            </div>
   313→            <div className="flex flex-wrap gap-2">
   314→              <Button variant="outline" onClick={exportCSV}><FileDown className="h-4 w-4 mr-2" /> CSV</Button>
   315→              <Button variant="outline" onClick={exportExcel}><FileSpreadsheet className="h-4 w-4 mr-2" /> Excel</Button>
   316→              <Button onClick={exportPDF}><FileText className="h-4 w-4 mr-2" /> PDF</Button>
   317→            </div>
   318→          </div>
   319→        </CardContent>
   320→      </Card>
   321→
   322→      {/* Summary cards */}
   323→      {loading ? (
   324→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   325→          {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24" />)}
   326→        </div>
   327→      ) : (
   328→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   329→          <SummaryCard label="Chiffre d'affaires" value={totalCA} />
   330→          <SummaryCard label="Achats" value={totalPurchases} />
   331→          <SummaryCard label="Frais plateforme" value={totalPlatformFees} />
   332→          <SummaryCard label="Frais de port" value={totalShipping} />
   333→          <SummaryCard label="Autres dépenses" value={totalOtherExpenses} />
   334→          <SummaryCard label="Bénéfice net" value={totalProfit} highlight />
   335→          <SummaryCard label="Marge nette" value={totalCA > 0 ? parseFloat(((totalProfit / totalCA) * 100).toFixed(1)) : 0} suffix="%" />
   336→          <SummaryCard label="Nb ventes" value={yearSales.length} />
   337→        </div>
   338→      )}
   339→
   340→      {/* Dépenses */}
   341→      <Card>
   342→        <CardHeader>
   343→          <div className="flex items-center justify-between">
   344→            <div>
   345→              <CardTitle className="text-base">Dépenses ({yearExpenses.length})</CardTitle>
   346→              <CardDescription>Abonnements, fournitures, carburant, etc.</CardDescription>
   347→            </div>
   348→            <Button size="sm" onClick={() => setShowForm(true)}>
   349→              <Plus className="h-4 w-4 mr-2" /> Ajouter
   350→            </Button>
   351→          </div>
   352→        </CardHeader>
   353→        <CardContent>
   354→          {yearExpenses.length === 0 ? (
   355→            <p className="text-sm text-muted-foreground py-8 text-center">Aucune dépense enregistrée pour {year}</p>
   356→          ) : (
   357→            <div className="overflow-x-auto">
   358→              <table className="w-full text-sm">
   359→                <thead>
   360→                  <tr className="text-left text-xs text-muted-foreground border-b">
   361→                    <th className="px-2 py-2 font-medium">Date</th>
   362→                    <th className="px-2 py-2 font-medium">Catégorie</th>
   363→                    <th className="px-2 py-2 font-medium">Libellé</th>
   364→                    <th className="px-2 py-2 font-medium text-right">Montant</th>
   365→                    <th className="px-2 py-2"></th>
   366→                  </tr>
   367→                </thead>
   368→                <tbody>
   369→                  {yearExpenses.map(e => (
   370→                    <tr key={e.id} className="border-b last:border-0 hover:bg-muted/30">
   371→                      <td className="px-2 py-2.5 text-xs text-muted-foreground">{formatDate(e.date)}</td>
   372→                      <td className="px-2 py-2.5">
   373→                        <Badge variant="secondary" className="text-[10px]">
   374→                          {EXPENSE_CATEGORIES.find(c => c.id === e.category)?.label || e.category}
   375→                        </Badge>
   376→                      </td>
   377→                      <td className="px-2 py-2.5">{e.label}</td>
   378→                      <td className="px-2 py-2.5 text-right font-medium">{formatEUR(e.amount)}</td>
   379→                      <td className="px-2 py-2.5 text-right">
   380→                        <Button size="sm" variant="ghost" className="h-7 w-7 p-0 text-rose-600" onClick={() => deleteExpense(e.id)}>
   381→                          <Trash2 className="h-3 w-3" />
   382→                        </Button>
   383→                      </td>
   384→                    </tr>
   385→                  ))}
   386→                </tbody>
   387→              </table>
   388→            </div>
   389→          )}
   390→        </CardContent>
   391→      </Card>
   392→
   393→      {/* Form dialog */}
   394→      {showForm && (
   395→        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4" onClick={() => setShowForm(false)}>
   396→          <Card className="max-w-md w-full" onClick={e => e.stopPropagation()}>
   397→            <CardHeader><CardTitle className="text-base">Nouvelle dépense</CardTitle></CardHeader>
   398→            <CardContent className="space-y-3">
   399→              <div className="grid grid-cols-2 gap-3">
   400→                <div className="space-y-1.5">
   401→                  <Label className="text-xs">Date</Label>
   402→                  <Input type="date" value={form.date} onChange={e => setForm({ ...form, date: e.target.value })} />
   403→                </div>
   404→                <div className="space-y-1.5">
   405→                  <Label className="text-xs">Catégorie</Label>
   406→                  <Select value={form.category} onValueChange={v => setForm({ ...form, category: v })}>
   407→                    <SelectTrigger><SelectValue /></SelectTrigger>
   408→                    <SelectContent>
   409→                      {EXPENSE_CATEGORIES.map(c => <SelectItem key={c.id} value={c.id}>{c.label}</SelectItem>)}
   410→                    </SelectContent>
   411→                  </Select>
   412→                </div>
   413→              </div>
   414→              <div className="space-y-1.5">
   415→                <Label className="text-xs">Libellé</Label>
   416→                <Input value={form.label} onChange={e => setForm({ ...form, label: e.target.value })} placeholder="Abonnement Vinted Pro" />
   417→              </div>
   418→              <div className="space-y-1.5">
   419→                <Label className="text-xs">Montant (€)</Label>
   420→                <Input type="number" step="0.01" value={form.amount} onChange={e => setForm({ ...form, amount: e.target.value })} placeholder="9.99" />
   421→              </div>
   422→              <div className="flex gap-2 pt-2">
